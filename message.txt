-- RemoteFunction:InvokeServer never times out by itself. A call the server doesn't answer
-- (maxed upgrade, not enough currency) would freeze the loop that made it, so wave buys go through this.
local function InvokeTimed(timeout, ...)
	local args = table.pack(...)
	local done, result = false, nil
	task.spawn(function()
		local ok, r = pcall(function() return getRemote:InvokeServer(table.unpack(args, 1, args.n)) end)
		result = ok and r or nil
		done = true
	end)
	local t0 = os.clock()
	while not done and os.clock() - t0 < timeout do task.wait() end
	return done, result
end

local function PostWebhook(url, payload)
	if not url or url == "" then return false end
	local success = false
	task.spawn(function()
		pcall(function()
			local body = HttpService:JSONEncode(payload)
			local headers = {["Content-Type"] = "application/json"}
			if request then
				request({Url = url, Method = "POST", Headers = headers, Body = body})
				success = true
			elseif http_request then
				http_request({Url = url, Method = "POST", Headers = headers, Body = body})
				success = true
			elseif syn and syn.request then
				syn.request({Url = url, Method = "POST", Headers = headers, Body = body})
				success = true
			end
		end)
	end)
	return success
end

local function GetBanStatus()
	if not isLobby then
		if isfile(CFG_BAN_CACHE) then
			getgenv().KiraHub.State.BanStatus = readfile(CFG_BAN_CACHE)
		else
			getgenv().KiraHub.State.BanStatus = "Lobby Only"
		end
		return getgenv().KiraHub.State.BanStatus
	end

	local bl = player:GetAttribute("Blacklisted") == true
	local ex = player:GetAttribute("Exploiter") == true
	if bl and ex then
		getgenv().KiraHub.State.BanStatus = "Banned"
	elseif bl or ex then
		getgenv().KiraHub.State.BanStatus = "Flagged"
	else
		getgenv().KiraHub.State.BanStatus = "Clean"
	end
	pcall(writefile, CFG_BAN_CACHE, getgenv().KiraHub.State.BanStatus)
	return getgenv().KiraHub.State.BanStatus
end

local function ScrapeLobbyStats()
	pcall(function()
		local lv = player:GetAttribute("Level")
		if lv then getgenv().KiraHub.State.LobbyLevel = tostring(lv) end
		local pr = player:GetAttribute("Prestige")
		getgenv().KiraHub.State.LobbyPrestige = pr and tostring(pr) or "0"
		local st = player:GetAttribute("Streak")
		if st then getgenv().KiraHub.State.LobbyStreak = tostring(st) end
	end)
	pcall(function()
		local topbar = INTERFACE:FindFirstChild("Topbar")
		if not topbar then return end
		local curr = topbar.Main:FindFirstChild("Currencies")
		if not curr then return end
		if curr:FindFirstChild("Gold") and curr.Gold:FindFirstChild("Amount") then
			getgenv().KiraHub.State.LobbyGold = curr.Gold.Amount.Text
		end
		if curr:FindFirstChild("Gems") and curr.Gems:FindFirstChild("Amount") then
			getgenv().KiraHub.State.LobbyGems = curr.Gems.Amount.Text
		end
	end)
end

local function CleanRewardName(rawName)
	if not rawName then return nil end
	local s = tostring(rawName)
	s = s:gsub("[%z\1-\31\127]", "")
	s = s:gsub("_", " ")
	s = s:match("^%s*(.-)%s*$") or s
	if s == "" or s:match("^[0-9%s]+$") then return nil end
	return s
end

local function FormatMissionTime(seconds)
	if not seconds or seconds < 0 then return "N/A" end
	local mins = math.floor(seconds / 60)
	local secs = math.floor(seconds % 60)
	return string.format("%02d:%02d", mins, secs)
end

-- Smooth Hub network queue: attack remotes are queued and sent 20 per frame (at most 400 waiting),
-- so multi hit and boss hit loops never burst past the game's rate limit.
;(function()
	local NET_BATCH, NET_QUEUE_MAX = 20, 400
	local queue, head, tail = {}, 1, 0
	function AutoFarm.NetPending() return tail - head + 1 end
	function AutoFarm.NetPush(remote, ...)
		tail += 1
		queue[tail] = {remote = remote, args = table.pack(...)}
	end
	-- one Slash + Register pair per nape (Smooth Hub's Slash)
	function AutoFarm.Slash(parts)
		if AutoFarm.NetPending() > NET_QUEUE_MAX then return false end
		for _, part in ipairs(parts) do
			AutoFarm.NetPush(postRemote, "Attacks", "Slash", true)
			AutoFarm.NetPush(postRemote, "Hitboxes", "Register", part, math.random(700, 900))
		end
		return true
	end
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not getgenv().KiraHubAOTR then conn:Disconnect() return end
		local sent = 0
		while head <= tail and sent < NET_BATCH do
			local item = queue[head]
			queue[head] = nil
			head += 1
			sent += 1
			if item.remote and item.remote.Parent then
				pcall(item.remote.FireServer, item.remote, table.unpack(item.args, 1, item.args.n))
			end
		end
		if head > tail then head, tail = 1, 0 end
	end)
end)()

function AutoFarm:Start()
	if self._running then return end
	if isLobby or isMenu then return end
	self._running = true
	-- each Start gets its own generation, so a Stop/Start (respawn) never leaves an old loop running
	local gen = (self._gen or 0) + 1
	self._gen = gen
	getgenv().KiraHub.State.MissionStartTime = os.clock()

	task.spawn(function()
		while self._running and self._gen == gen do
			local char = player.Character
			if char then
				local hrp = char:FindFirstChild("HumanoidRootPart")
				if hrp then hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0) end
			end
			task.wait()
		end
	end)

	task.spawn(function()
		local function isWavesMode()
			return workspace:FindFirstChild("Waves") ~= nil
		end

		-- Smooth Hub's map gates: game loaded, character alive and not frozen, titans with a nape
		local function checkReady()
			if not game:IsLoaded() then return false, "game loading" end
			local char = player.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if not hrp then return false, "character loading" end
			if hrp.Anchored then return false, "character frozen" end
			local titans = workspace:FindFirstChild("Titans")
			if titans then
				for _, v in ipairs(titans:GetChildren()) do
					local hb = v:FindFirstChild("Hitboxes")
					local hit = hb and hb:FindFirstChild("Hit")
					if hit and hit:FindFirstChild("Nape") then return true end
				end
			end
			return false, "titans not spawned"
		end

		while self._running and self._gen == gen do
			local ready, why = checkReady()
			if ready then break end
			AutoFarm.SetStatus("Waiting for map to load (" .. why .. ")")
			task.wait(1)
		end
		if not self._running then return end

		local antiDetection = GetOption("AntiDetectionWaitSlider", 0)
		if antiDetection > 0 then task.wait(antiDetection) end

		if IsToggled("WaitBeforeFarmingToggle") then
			local waitSecs = GetOption("WaitBeforeFarmingSlider", 15)
			task.wait(waitSecs)
		end
		if not self._running then return end

		local titansFolder = workspace:FindFirstChild("Titans")
		local lastAttack = 0
		local currentChar, root = nil, nil
		local attackTitanSpawnTime = nil

		local function updateCharState()
			local char = player.Character
			if not char then return false end
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if not hrp then return false end
			if char ~= currentChar then
				currentChar = char
				root = hrp
			end
			return true
		end

		local validNapes = {}
		local nextTitanCacheUpdate = 0
		local nextObjectiveCacheUpdate = 0
		local cachedObjectivePart = nil
		local masteryComboIndex = 1
		local nextMapDataTry = 0
		local lastMasteryPunch = 0

		while self._running and self._gen == gen do
			local wavesMode = workspace:FindFirstChild("Waves") ~= nil

			if getgenv().KiraHub.State.EmergencyStop then
				task.wait(0.1)
				continue
			end

			-- Auto Shop / Auto Roll Blessings are at an NPC: hold the farm until they're back
			if getgenv().KiraHub.State.NpcTrip then
				AutoFarm.SetStatus("Paused (" .. tostring(getgenv().KiraHub.State.NpcTrip) .. " trip)")
				task.wait(0.1)
				continue
			end

			if IsToggled("SkinHunterToggle") and not isLobby then
				task.wait(1)
				continue
			end

			if IsToggled("FailSafeToggle") and getgenv().KiraHub.State.MissionStartTime then
				local elapsed = os.clock() - getgenv().KiraHub.State.MissionStartTime
				local limitSecs = GetOption("FailsafeMinutesSlider", 15) * 60
				if elapsed >= limitSecs then
					self:Stop()
					pcall(function() getRemote:InvokeServer("Functions", "Teleport", "Lobby") end)
					task.wait(0.5)
					pcall(function() TeleportService:Teleport(LOBBY_ID, player) end)
					break
				end
			end

			if player:GetAttribute("Cutscene") then task.wait(0.05) continue end

			-- mission data only adds weapon/mastery details: refresh it in the background with a
			-- timeout and keep farming without it (an unanswered InvokeServer used to freeze this loop)
			if not wavesMode and not mapData and os.clock() >= nextMapDataTry then
				nextMapDataTry = os.clock() + 5
				task.spawn(function()
					local _, data = InvokeTimed(5, "Data", "Copy")
					if data then mapData = data end
				end)
			end

			local slotIndex = player:GetAttribute("Slot")
			local slotData = slotIndex and mapData and mapData.Slots
				and (mapData.Slots[slotIndex] or mapData.Slots[tostring(slotIndex)])

			if IsToggled("DieAtStreakToggle") then
				local streak = player:GetAttribute("Streak") or 0
				local threshold = GetOption("DieAtStreakSlider", 10000)
				if streak >= threshold then
					local char = player.Character
					if char then
						local hum = char:FindFirstChildOfClass("Humanoid")
						if hum then hum.Health = 0 end
					end
				end
			end

			local isBlades = true
			if slotData then
				isBlades = slotData.Weapon == "Blades"
			else
				local wt = GetCurrentWeapon()
				isBlades = wt ~= "Spears"
			end
			getgenv().KiraHub.State.AttackCooldownVal = isBlades and 0.15 or 1

			local playerCount = workspace:GetAttribute("Player_Count") or #Players:GetPlayers()
			if IsToggled("SoloOnlyToggle") and playerCount > 1 then
				self:Stop()
				pcall(function() getRemote:InvokeServer("Functions", "Teleport", "Lobby") end)
				task.wait(0.5)
				TeleportService:Teleport(LOBBY_ID, player)
				break
			end

			if not updateCharState() then task.wait(0.05) continue end
			titansFolder = workspace:FindFirstChild("Titans") or titansFolder

			local ws_ObjectiveFolder = workspace:FindFirstChild("Unclimbable")
				and workspace.Unclimbable:FindFirstChild("Objective")
			local rs_ObjectiveFolder = ReplicatedStorage:FindFirstChild("Objectives")
			local mapType = wavesMode and "Waves"
				or (workspace:GetAttribute("Type") or (mapData and mapData.Map and mapData.Map.Type))

			local isArmoredRaid = not wavesMode and ws_ObjectiveFolder and ws_ObjectiveFolder:FindFirstChild("Armored_Boss")
			local isFemaleRaid = not wavesMode and rs_ObjectiveFolder and rs_ObjectiveFolder:FindFirstChild("Defeat_Annie")
			local femaleExists = not wavesMode and ws_ObjectiveFolder and ws_ObjectiveFolder:FindFirstChild("Female_Boss")
			local attackExists = not wavesMode and ws_ObjectiveFolder and ws_ObjectiveFolder:FindFirstChild("Attack_Boss")
			local armoredTitan = not wavesMode and titansFolder and titansFolder:FindFirstChild("Armored_Titan")
			local hasReinerObjective = armoredTitan and armoredTitan:GetAttribute("State")

			local isColossalRaid = not wavesMode and (
				(rs_ObjectiveFolder and rs_ObjectiveFolder:FindFirstChild("Defeat_Bertholdt"))
				or (ws_ObjectiveFolder and ws_ObjectiveFolder:FindFirstChild("Colossal_Boss"))
			)

			local isDefendEren = not wavesMode and rs_ObjectiveFolder and rs_ObjectiveFolder:FindFirstChild("Defend_Eren")
			local isDefendBoats = not wavesMode and rs_ObjectiveFolder and rs_ObjectiveFolder:FindFirstChild("Defend_Boats")
			local isRaidDefensePhase = (isDefendEren or isDefendBoats) and IsToggled("AutoFarmRaidsToggle")

			if not wavesMode and isFemaleRaid and not femaleExists and not attackExists then
				task.wait(0.05)
				continue
			end

			if not wavesMode and isColossalRaid then
				local stallObjective = rs_ObjectiveFolder and rs_ObjectiveFolder:FindFirstChild("Stall_Colossal_Titan")
				local stallDone = stallObjective and stallObjective.Value >= (stallObjective:GetAttribute("Requirement") or 1)
				if not stallDone then
					local walls = workspace:FindFirstChild("Climbable") and workspace.Climbable:FindFirstChild("Walls")
					local cannonModel = nil
					if walls then
						cannonModel = walls:FindFirstChild("Wall") and walls.Wall:FindFirstChild("Cannons") and walls.Wall.Cannons:FindFirstChild("1")
						if not cannonModel then
							for _, wall in ipairs(walls:GetChildren()) do
								local c = wall:FindFirstChild("Cannons") and wall.Cannons:FindFirstChild("1")
								if c then cannonModel = c break end
							end
						end
					end
					if cannonModel and not self._colossalCannon then
						self._colossalCannon = true
						task.spawn(function()
							local function getLiveNapePos()
								local tf = workspace:FindFirstChild("Titans")
								local ct = tf and tf:FindFirstChild("Colossal_Titan")
								if not ct then return nil end
								local hit = ct:FindFirstChild("Hitboxes") and ct.Hitboxes:FindFirstChild("Hit")
								if hit and hit:FindFirstChild("Nape") then return hit.Nape.Position end
								local fake = ct:FindFirstChild("Fake")
								local head = fake and fake:FindFirstChild("Head")
								return head and head.Position or nil
							end
							local function isStallComplete()
								local so = ReplicatedStorage:FindFirstChild("Objectives")
									and ReplicatedStorage.Objectives:FindFirstChild("Stall_Colossal_Titan")
								return so and so.Value >= (so:GetAttribute("Requirement") or 1)
							end
							local impactConn = postRemote.OnClientEvent:Connect(function(a1, a2, cannonObj)
								if a1 ~= "Skills" or a2 ~= "Impact" then return end
								if not self._colossalCannon then return end
								local ref = cannonObj or workspace:FindFirstChild("Cannon")
								if not ref then return end
								for i = 1, 30 do
									local napePos = getLiveNapePos()
									if napePos then
										pcall(function() postRemote:FireServer("S_Skills", "Impact", ref, napePos) end)
									end
								end
							end)
							while self._running and not isStallComplete() do
								pcall(function() getRemote:InvokeServer("Cannon", "State", cannonModel, true) end)
								task.wait(0.3)
								for shootAttempt = 1, 3 do
									local shootOk = pcall(function()
										getRemote:InvokeServer("Cannon", "Shoot", {BarrelWood = 40, Base = 0})
									end)
									if shootOk then break end
									task.wait(0.1)
								end
								task.wait(1.8)
							end
							impactConn:Disconnect()
							pcall(function() getRemote:InvokeServer("Cannon", "State", cannonModel, false) end)
							self._colossalCannon = false
						end)
					end
					task.wait(0.05)
					continue
				end
			end

			if math.fmod(os.clock(), 0.5) < 0.05 then
				pcall(function()
					if root then root.CanCollide = false end
				end)
			end

			local now = os.clock()
			local isShifted = currentChar and currentChar:GetAttribute("Shifter") or false

			if IsToggled("MasteryFarmToggle") then
				local shiftReady = player:GetAttribute("Bar") and player:GetAttribute("Bar") == 100
				if not isShifted and shiftReady then
					repeat
						pcall(function() getRemote:InvokeServer("S_Skills", "Usage", "999", false) end)
						task.wait(1)
					until not self._running or (player.Character and player.Character:GetAttribute("Shifter"))
					continue
				end
			end

			if now >= nextTitanCacheUpdate then
				nextTitanCacheUpdate = now + 0.1
				table.clear(validNapes)
				if titansFolder then
					for _, v in ipairs(titansFolder:GetChildren()) do
						if v:GetAttribute("Killed") then continue end
						local hit = v:FindFirstChild("Hitboxes") and v.Hitboxes:FindFirstChild("Hit")
						if hit then
							local fake = v:FindFirstChild("Fake")
							if fake and fake:FindFirstChild("Collision") and not fake.Collision.CanCollide then continue end
							local nape = hit:FindFirstChild("Nape")
							if nape then
								if IsToggled("AutoIceburstToggle") then
									local isIceburst = v:GetAttribute("Type") == "Iceburst" or v.Name:find("Iceburst")
									if isIceburst then table.insert(validNapes, 1, nape)
									else table.insert(validNapes, nape) end
								else
									table.insert(validNapes, nape)
								end
							end
						end
					end
				end
			end

			if #validNapes == 0 then
				AutoFarm.SetStatus("Searching for titans")
				if root then root.AssemblyLinearVelocity = Vector3.new(0, 0, 0) end
				task.wait(0.05)
				continue
			end

			local rootPos = root.Position
			local referencePos = rootPos
			local objectiveFound = false

			if not wavesMode and now >= nextObjectiveCacheUpdate then
				nextObjectiveCacheUpdate = now + 1
				cachedObjectivePart = nil
				if ws_ObjectiveFolder then
					for _, desc in ipairs(ws_ObjectiveFolder:GetDescendants()) do
						if desc:IsA("BillboardGui") and desc.Parent and desc.Parent:IsA("BasePart") then
							cachedObjectivePart = desc.Parent
							break
						end
					end
				end
			end

			if not wavesMode and cachedObjectivePart and cachedObjectivePart.Parent then
				referencePos = cachedObjectivePart.Position
				objectiveFound = true
			end

			if isRaidDefensePhase and objectiveFound then
				referencePos = cachedObjectivePart.Position
			end

			local useRangeLimit = not wavesMode and objectiveFound and isArmoredRaid and not hasReinerObjective
			local closestDist, closestNape = math.huge, nil
			local closestIsBoss = false
			local bossDist, bossHitPoint = math.huge, nil
			local attackTitanFound = false
			local highestZ = -math.huge
			local isStall = not wavesMode and mapData and mapData.Map and mapData.Map.Objective == "Stall"
			local bossIsRoaring = false

			for i = 1, #validNapes do
				local nape = validNapes[i]
				if not nape or not nape.Parent then continue end
				local titanModel = nape.Parent.Parent.Parent
				local fake = titanModel:FindFirstChild("Fake")
				if (fake and fake:FindFirstChild("Collision") and not fake.Collision.CanCollide)
					or titanModel:GetAttribute("Dead") then continue end
				local tName = titanModel.Name
				local isBoss = getgenv().KiraHub.Data.BossNames[tName]
				if not wavesMode and isArmoredRaid and not hasReinerObjective and tName == "Armored_Titan" then continue end
				if not wavesMode and isBoss and not titanModel:GetAttribute("State") then continue end
				local isRoaring = not wavesMode and isBoss and (
					titanModel:GetAttribute("Attack") == "Roar"
					or titanModel:GetAttribute("Attack") == "Berserk_Mode"
				)
				if tName == "Attack_Titan" then attackTitanFound = true end
				local dx = referencePos.X - nape.Position.X
				local dz = referencePos.Z - nape.Position.Z
				local d = dx * dx + dz * dz
				local adjustedDist = d
				if getgenv().KiraHub.State.TargetNape == nape then adjustedDist = adjustedDist - 15000 end
				if useRangeLimit and d > 90000 then continue end
				if not wavesMode and isBoss then
					local hitPart = (titanModel:FindFirstChild("Marker") and titanModel.Marker.Adornee)
						or titanModel.Hitboxes.Hit.Nape
					if hitPart and adjustedDist < bossDist then
						bossDist = adjustedDist
						bossHitPoint = hitPart
						bossIsRoaring = isRoaring
					end
				end
				if isStall then
					if nape.Position.Z > highestZ then highestZ = nape.Position.Z closestNape = nape end
				elseif adjustedDist < closestDist then
					closestDist = adjustedDist
					closestNape = nape
					closestIsBoss = isBoss
				end
			end

			local targetPart, targetIsRoaring = nil, false

			if wavesMode then
				targetPart = closestNape
			elseif isRaidDefensePhase then
				targetPart = closestNape or bossHitPoint
				targetIsRoaring = (targetPart == bossHitPoint) and bossIsRoaring or false
			else
				local prio = GetOption("RaidPriorityDropdown", "Boss First")
				if prio == "Boss First" then
					targetPart = bossHitPoint or closestNape
					targetIsRoaring = (targetPart == bossHitPoint) and bossIsRoaring or false
				elseif prio == "Adds First" then
					targetPart = closestNape or bossHitPoint
					targetIsRoaring = (targetPart == bossHitPoint) and bossIsRoaring or false
				elseif prio == "Mixed" then
					if math.random(1, 3) == 1 and bossHitPoint then
						targetPart = bossHitPoint
						targetIsRoaring = bossIsRoaring
					else
						targetPart = closestNape or bossHitPoint
						targetIsRoaring = (targetPart == bossHitPoint) and bossIsRoaring or false
					end
				else
					targetPart = closestNape
					if targetPart == bossHitPoint then targetIsRoaring = bossIsRoaring end
				end
			end

			if useRangeLimit and closestNape then
				targetPart = closestNape
				targetIsRoaring = false
			end

			if not getgenv().KiraHub.State._missionStartClock then getgenv().KiraHub.State._missionStartClock = os.clock() end

			if not wavesMode and targetPart and #validNapes == 1 and mapType == "Missions" then
				local elapsed = os.clock() - getgenv().KiraHub.State._missionStartClock
				local requiredWait = IsToggled("LastTitanWaitToggle") and GetOption("LastTitanWaitSlider", 60) or 10
				local safeBanTime = GetOption("SafeBanTimeSlider", 0)
				if safeBanTime > 0 and elapsed < safeBanTime then
					targetPart = nil
				elseif elapsed < requiredWait then
					targetPart = nil
				end
			end

			getgenv().KiraHub.State.TargetNape = targetPart
			if not targetPart then
				AutoFarm.SetStatus("Holding (last titan wait / boss not active yet)")
			elseif targetIsRoaring then
				AutoFarm.SetStatus("Waiting (boss is roaring)")
			else
				AutoFarm.SetStatus("Farming")
			end
			if attackTitanFound then
				attackTitanSpawnTime = attackTitanSpawnTime or now
			else
				attackTitanSpawnTime = nil
			end
			local attackTitanReady = not attackTitanFound
				or (attackTitanSpawnTime and (now - attackTitanSpawnTime) >= 5)

			if targetPart then
				local currentTitanModel = targetPart
				while currentTitanModel and currentTitanModel.Parent ~= titansFolder do
					currentTitanModel = currentTitanModel.Parent
				end

				if isShifted then
					local targetHRP = currentTitanModel and currentTitanModel:FindFirstChild("HumanoidRootPart")
					local targetCFrame = targetHRP and targetHRP.CFrame or targetPart.CFrame
					root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
					root.CFrame = targetCFrame * CFrame.new(0, 0, 80)
					local mode = GetOption("MasteryModeDropdown", "Both")
					local doPunch = mode == "Punching" or mode == "Both"
					local doSkills = mode == "Skill Usage" or mode == "Both"
					if not targetIsRoaring then
						if doPunch and (now - lastMasteryPunch) >= 1 then
							lastMasteryPunch = now
							postRemote:FireServer("Attacks", "Slash", true)
							postRemote:FireServer("Hitboxes", "Register", targetPart, nil, nil, masteryComboIndex)
							masteryComboIndex = masteryComboIndex + 1
							if masteryComboIndex > 4 then masteryComboIndex = 1 end
						end
						if doSkills and slotData and slotData.Skills and slotData.Skills.Shifter
							and not getgenv().KiraHub.State.ShifterSkillsRunning
						then
							getgenv().KiraHub.State.ShifterSkillsRunning = true
							task.spawn(function()
								for _, skillId in ipairs(slotData.Skills.Shifter) do
									local idNum = tonumber(skillId)
									if idNum and idNum ~= 200 and idNum ~= 300 and idNum ~= 400
										and idNum ~= 210 and idNum ~= 211 and idNum ~= 306
										and idNum ~= 308 and idNum ~= 402 and idNum ~= 403 and idNum ~= 407
									then
										pcall(function() getRemote:InvokeServer("S_Skills", "Usage", tostring(skillId), false) end)
									end
									task.wait(1)
								end
								getgenv().KiraHub.State.ShifterSkillsRunning = false
							end)
						end
					end
					task.wait(0.05)
					continue
				end

				local titanHRP = currentTitanModel and currentTitanModel:FindFirstChild("HumanoidRootPart")
				local floatHeight = GetOption("FloatHeightSlider", 250)
				local targetHeightPos
				if titanHRP then
					targetHeightPos = (titanHRP.CFrame * CFrame.new(0, floatHeight, 30)).Position
				else
					targetHeightPos = targetPart.Position + Vector3.new(0, floatHeight, 0)
				end

				local moveMode = GetOption("MovementModeDropdown", "Hover")
				local moveSpeed = GetOption("HoverSpeedSlider", 400)
				if moveMode == "Hover" then
					local distToTarget = (targetHeightPos - rootPos).Magnitude
					if distToTarget < 20 then
						local titanBasePos = titanHRP and titanHRP.Position or targetPart.Position
						getgenv().KiraHub.State.OrbitAngle = (getgenv().KiraHub.State.OrbitAngle or 0) + 0.18
						if getgenv().KiraHub.State.OrbitAngle > math.pi * 2 then
							getgenv().KiraHub.State.OrbitAngle = getgenv().KiraHub.State.OrbitAngle - math.pi * 2
						end
						local orbitRadius = 15
						local ang = getgenv().KiraHub.State.OrbitAngle
						local orbitX = titanBasePos.X + math.cos(ang) * orbitRadius
						local orbitZ = titanBasePos.Z + math.sin(ang) * orbitRadius
						local orbitPos = Vector3.new(orbitX, targetHeightPos.Y, orbitZ)
						root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
						root.CFrame = CFrame.new(orbitPos, Vector3.new(titanBasePos.X, orbitPos.Y, titanBasePos.Z))
					else
						local dir = targetHeightPos - rootPos
						if dir.Magnitude > 1 then
							root.AssemblyLinearVelocity = dir.Unit * moveSpeed
						else
							root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						end
					end
				elseif moveMode == "Teleport" then
					root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
					root.CFrame = CFrame.new(targetHeightPos)
				elseif moveMode == "Tween Farm" then
					-- Smooth Hub tweening: this loop only aims; the per-frame driver after AutoFarm:Stop() moves the root
					local st = getgenv().KiraHub.State
					st.TweenGoal = targetPart.Position + Vector3.new(0, math.max(floatHeight, 0), 0)
					st.TweenLook = targetPart.Position
					st.TweenSpeed = moveSpeed
					st.TweenGoalAt = os.clock()
				end

				if not attackTitanReady then task.wait(0.05) continue end

				local dx = root.Position.X - targetPart.Position.X
				local dz = root.Position.Z - targetPart.Position.Z
				local attackRange = GetOption("AttackRangeSlider", 150)
				local AttackRangeSq = attackRange * attackRange

				if not targetIsRoaring
					and (dx * dx + dz * dz) <= AttackRangeSq
					and (now - lastAttack) >= getgenv().KiraHub.State.AttackCooldownVal
				then
					lastAttack = now

					local hitTargets = {targetPart}
					if IsToggled("MultiHitToggle") then
						-- Smooth Hub multi hit: the main target first, then the closest other living napes
						local maxHits = GetOption("MultiHitCountSlider", 3)
						local others = {}
						for i = 1, #validNapes do
							local n = validNapes[i]
							if n ~= targetPart and n and n.Parent then
								local titan = n.Parent.Parent and n.Parent.Parent.Parent
								local hum = titan and titan:FindFirstChildOfClass("Humanoid")
								if titan and not titan:GetAttribute("Dead") and not titan:GetAttribute("Killed")
									and (not hum or hum.Health > 0)
								then
									others[#others + 1] = {part = n, d = (n.Position - targetPart.Position).Magnitude}
								end
							end
						end
						table.sort(others, function(a, b) return a.d < b.d end)
						for i = 1, math.min(maxHits - 1, #others) do
							hitTargets[#hitTargets + 1] = others[i].part
						end
					end

					if isBlades then
						-- Smooth Hub Slash through the rate-limited queue; bosses get Boss Hits loops
						local thisBoss = IsBossTarget(targetPart)
						local bladeLoops = thisBoss and GetOption("BladeBossHitsSlider", 35) or 1
						for _ = 1, bladeLoops do
							if not AutoFarm.Slash(hitTargets) then break end
						end
					else
						local spearsLabel = nil
						pcall(function()
							spearsLabel = pGui.Interface.HUD.Main.Top["7"].Spears.Spears
						end)
						local currentAmmo = 0
						if spearsLabel then
							local cur = string.match(spearsLabel.Text, "(%d+)%s*/")
							currentAmmo = tonumber(cur) or 0
						end
						if currentAmmo > 0 then
							task.spawn(function()
								pcall(function() getRemote:InvokeServer("Spears", "S_Fire", tostring(currentAmmo)) end)
								local isBoss = IsBossTarget(targetPart)
								local loops = isBoss and (GetOption("BladeBossHitsSlider") or 35) or 1
								for j = 1, loops do
									for _, nape in ipairs(hitTargets) do
										pcall(function() postRemote:FireServer("Spears", "S_Explode", nape.Position) end)
									end
								end
							end)
						else
							local station = GetRefillStation()
							pcall(function() postRemote:FireServer("Attacks", "Reload", station) end)
						end
					end
				end
			else
				if root then root.AssemblyLinearVelocity = Vector3.new(0, 0, 0) end
			end

			task.wait(0.05)
		end
	end)
end

function AutoFarm:Stop()
	self._running = false
	self._lastStatus = nil
	self._colossalCannon = false
	getgenv().KiraHub.State.MissionStartTime = nil
	getgenv().KiraHub.State.TweenGoal = nil
end

-- Smooth Hub tween driver (Tween Farm mode): every frame, step the root toward the goal at
-- Movement Speed per second, orbiting the nape, and face the nape once within arrive range.
-- A goal older than 0.3s means the farm loop stopped aiming (roar, boss wait, no target), so it holds still.
do
	local ORBIT_RADIUS, ORBIT_SPEED, ORBIT_BOB, ARRIVE_DIST = 14, 1.6, 5, 12
	local orbitAngle = 0
	local conn
	conn = RunService.Heartbeat:Connect(function(dt)
		if not getgenv().KiraHubAOTR then conn:Disconnect() return end
		local st = getgenv().KiraHub.State
		if not AutoFarm._running or st.NpcTrip or not st.TweenGoal or os.clock() - (st.TweenGoalAt or 0) > 0.3 then return end
		if GetOption("MovementModeDropdown", "Hover") ~= "Tween Farm" then return end
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if not root then return end

		orbitAngle = (orbitAngle + ORBIT_SPEED * dt) % (math.pi * 2)
		local goal = st.TweenGoal + Vector3.new(
			math.cos(orbitAngle) * ORBIT_RADIUS,
			math.sin(orbitAngle * 2) * ORBIT_BOB,
			math.sin(orbitAngle) * ORBIT_RADIUS
		)
		local diff = goal - root.Position
		local dist = diff.Magnitude
		pcall(function() root.AssemblyLinearVelocity = Vector3.zero end)
		if dist > 1 then
			local step = math.min((st.TweenSpeed or 600) * dt, dist)
			local newPos = root.Position + diff.Unit * step
			if dist <= ARRIVE_DIST then
				local look = st.TweenLook
				root.CFrame = CFrame.lookAt(newPos, Vector3.new(look.X, newPos.Y, look.Z))
			else
				root.CFrame = CFrame.new(newPos)
			end
		end
	end)
end

local lastReloadTime = 0
local lastRefillTime = 0
local lastAmmoValueSeen = -1
local lastAmmoChangeAt = 0
local lastBladeValueSeen = -1
local lastBladeChangeAt = 0
local currentTrackedChar = nil

player.CharacterAdded:Connect(function()
	lastReloadTime = 0
	lastRefillTime = 0
	lastAmmoValueSeen = -1
	lastAmmoChangeAt = os.clock()
	lastBladeValueSeen = -1
	lastBladeChangeAt = os.clock()
	currentTrackedChar = nil
end)

task.spawn(function()
	while task.wait(0.3) do
		if not getgenv().KiraHubAOTR then break end
		if not IsToggled("AutoReloadToggle") then continue end
		if isLobby or isMenu then continue end

		local char = player.Character
		if not char then continue end

		if char ~= currentTrackedChar then
			currentTrackedChar = char
			lastBladeValueSeen = -1
			lastBladeChangeAt = os.clock()
			lastReloadTime = 0
			lastRefillTime = 0
		end

		local hrp = char:FindFirstChild("HumanoidRootPart")
		if not hrp then continue end

		local weaponType = GetCurrentWeapon()
		if not weaponType then continue end
		local now = os.clock()
		local station = GetRefillStation()

		if weaponType == "Blades" then
			local current = GetBladeSetCount()
			if not current then continue end

			if current ~= lastBladeValueSeen then
				lastBladeValueSeen = current
				lastBladeChangeAt = now
			end

			local rig = char:FindFirstChild("Rig_" .. player.Name)
			local blade = rig and rig:FindFirstChild("LeftHand") and rig.LeftHand:FindFirstChild("Blade_1")
			local bladeBroken = blade and blade.Transparency == 1

			if current == 0 and (now - lastRefillTime) > 1.2 then
				lastRefillTime = now
				lastReloadTime = now
				pcall(function() postRemote:FireServer("Attacks", "Reload", station) end)
			elseif current > 0 and bladeBroken and (now - lastReloadTime) > 0.6 then
				lastReloadTime = now
				pcall(function() getRemote:InvokeServer("Blades", "Reload") end)
			end

		elseif weaponType == "Spears" then
			local count = GetSpearAmmoCount()
			if not count then continue end

			if count == 0 and (now - lastRefillTime) > 1.2 then
				lastRefillTime = now
				lastReloadTime = now
				pcall(function() postRemote:FireServer("Attacks", "Reload", station) end)
			end
		end
	end
end)

task.spawn(function()
	while task.wait(0.1) do
		if not getgenv().KiraHubAOTR then break end
		if not IsToggled("ManualAutoAttackToggle") then continue end
		if AutoFarm._running then continue end
		if getgenv().KiraHub.State.EmergencyStop then continue end
		if isLobby or isMenu then continue end
		local char = player.Character
		if not char then continue end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if not hrp then continue end
		local tf = workspace:FindFirstChild("Titans")
		if not tf then continue end
		local closestNape, closestDist = nil, 40
		for _, t in ipairs(tf:GetChildren()) do
			pcall(function()
				if t:GetAttribute("Killed") or t:GetAttribute("Dead") then return end
				local hum = t:FindFirstChildOfClass("Humanoid")
				if not hum or hum.Health <= 0 then return end
				local hit = t:FindFirstChild("Hitboxes") and t.Hitboxes:FindFirstChild("Hit")
				if not hit then return end
				local nape = hit:FindFirstChild("Nape")
				if nape then
					local dist = (hrp.Position - nape.Position).Magnitude
					if dist < closestDist then closestDist = dist closestNape = nape end
				end
			end)
		end
		if closestNape then
			postRemote:FireServer("Attacks", "Slash", true)
			postRemote:FireServer("Hitboxes", "Register", closestNape, math.random(625, 850))
			task.wait(0.5)
		end
	end
end)

local function ForceRetry()
	local rewardsGui = INTERFACE:FindFirstChild("Rewards")
	if rewardsGui and rewardsGui.Visible then
		local retryBtn = nil
		pcall(function() retryBtn = rewardsGui.Main.Info.Main.Buttons.Retry end)
		if not retryBtn or not retryBtn.Visible then
			for _, btn in ipairs(rewardsGui:GetDescendants()) do
				if (btn:IsA("TextButton") or btn:IsA("ImageButton")) and btn.Visible and btn.Active then
					if btn.Name == "Retry" or (btn:IsA("TextButton") and btn.Text:find("Retry")) then
						retryBtn = btn
						break
					end
				end
			end
		end
		if retryBtn and retryBtn.Visible and retryBtn.Active then
			UseButton(retryBtn)
			task.wait(0.2)
			if rewardsGui.Visible then
				GuiService.SelectedObject = retryBtn
				task.wait(0.1)
				VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
				VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
			end
			if rewardsGui.Visible then
				local pos = retryBtn.AbsolutePosition + retryBtn.AbsoluteSize / 2
				VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 0)
				task.wait(0.1)
				VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 0)
			end
			return true
		end
	end
	pcall(function() getRemote:InvokeServer("Functions", "Retry", "Add") end)
	task.wait(0.3)
	pcall(function() postRemote:FireServer("Functions", "Retry") end)
	return true
end

-- snapshot of every switch for Share Config (webhooks, keys, keybinds and theme are never included);
-- taken once before the autoload config so share codes only carry what you changed
getgenv().KiraHub.SnapshotOptions = function()
	local function skip(idx)
		local l = tostring(idx):lower()
		return l:find("webhook") or l:find("url") or l:find("key") or l:find("feedback")
			or l:find("share") or l:find("savemanager") or l:find("theme")
	end
	local snap = {t = {}, o = {}}
	for idx, tog in pairs(Toggles) do
		if not skip(idx) and type(tog) == "table" and type(tog.Value) == "boolean" then snap.t[idx] = tog.Value end
	end
	for idx, opt in pairs(Options) do
		if not skip(idx) and type(opt) == "table" then
			if opt.Type == "Slider" and type(opt.Value) == "number" then
				snap.o[idx] = opt.Value
			elseif opt.Type == "Input" and type(opt.Value) == "string" then
				snap.o[idx] = opt.Value
			elseif opt.Type == "Dropdown" then
				if opt.Multi then
					local list = {}
					for k, on in pairs(opt.Value or {}) do
						if on then list[#list + 1] = tostring(k) end
					end
					table.sort(list)
					snap.o[idx] = list
				elseif type(opt.Value) == "string" then
					snap.o[idx] = opt.Value
				end
			end
		end
	end
	return snap
end
pcall(function() getgenv().KiraHub.ConfigDefaults = getgenv().KiraHub.SnapshotOptions() end)

local function FormatStatsTable(tbl)
	local str = ""
	for k, v in pairs(tbl) do 
		str = str .. k .. ": " .. tostring(v) .. "\n" 
	end
	return str ~= "" and str or "None"
end

local function FormatItemsTable(tbl)
	local str = ""
	for name, qty in pairs(tbl) do
		str = str .. "[+] " .. name:gsub("_", " ") .. " (x" .. qty .. ")\n"
	end
	return str ~= "" and str or "None"
end

local function GetPerkXP(rarity, level)
	local baseXP = {Common = 100, Rare = 250, Epic = 625, Legendary = 2500, Secret = 10000}
	return (baseXP[rarity] or 0) * math.max(level, 1)
end

local function ScrapeMissionRewards()
	local data = {Stats = {}, Total = {}, Items = {}, Special = {}}
	local dropsData = {}
	
	if statsFrame then
		for _, v in ipairs(statsFrame:GetChildren()) do
			if v:IsA("Frame") and v:FindFirstChild("Stat") and v:FindFirstChild("Amount") then
				data.Stats[v.Name:gsub("_", " ")] = v.Amount.Text
			end
		end
	end
	
	pcall(function()
		local res = getRemote:InvokeServer("S_Rewards", "Get", true)
		if res and res.Obtained then
			local ob = res.Obtained
			if ob.Gold and ob.Gold > 0 then data.Items["Gold"] = tostring(ob.Gold) end
			if ob.XP and ob.XP > 0 then data.Items["XP"] = tostring(ob.XP) end
			if ob.Gems and ob.Gems > 0 then data.Items["Gems"] = tostring(ob.Gems) end
			if ob.Canes and ob.Canes > 0 then data.Items["Canes"] = tostring(ob.Canes) end
			if ob.Shards and ob.Shards > 0 then data.Items["Shards"] = tostring(ob.Shards) end
			if ob.Silver and ob.Silver > 0 then data.Items["Silver"] = tostring(ob.Silver) end
			if ob.BP_XP and ob.BP_XP > 0 then data.Items["BP XP"] = tostring(ob.BP_XP) end
			if ob.Perks then 
				for _, pn in ipairs(ob.Perks) do 
					local name = CleanRewardName(pn)
					data.Items["Perk: " .. name] = "1" 
				end 
			end
			if ob.Drops then
				for itemId, itemName in pairs(ob.Drops) do
					local name = CleanRewardName(itemName)
					dropsData[tostring(itemId)] = name
					local existing = tonumber(data.Items[name]) or 0
					data.Items[name] = tostring(existing + 1)
				end
			end
			if ob.Chests then
				for cn, cq in pairs(ob.Chests) do
					if cq and cq > 0 then data.Items["Chest: " .. cn] = tostring(cq) end
				end
			end
		end
	end)

	pcall(function()
		local res2 = getRemote:InvokeServer("S_Rewards", "Get", true)
		if res2 and res2.Obtained and res2.Obtained.Perks then
			for _, pn in ipairs(res2.Obtained.Perks) do
				local rarity = GetPerkRarity(pn)
				if rarity == "Secret" or rarity == "Legendary" then 
					local name = CleanRewardName(pn)
					data.Special["Perk: " .. name] = "1" 
				end
			end
		end
	end)

	if itemsFrame then
		for _, v in ipairs(itemsFrame:GetChildren()) do
			if v:IsA("Frame") and v:FindFirstChild("Main") then
				local inner = v.Main:FindFirstChild("Inner")
				if inner then
					local rarityFrame = inner:FindFirstChild("Rarity")
					local isSpecial = false
					if rarityFrame then
						local c = rarityFrame.BackgroundColor3
						-- Red (Mythical) or Gold (Legendary)
						if (math.abs(c.R - 1) < 0.05 and c.G < 0.1 and c.B < 0.1)
							or (c.R > 0.9 and c.G > 0.6 and c.B < 0.3)
						then
							isSpecial = true
						end
					end

					-- Also force-flag known serum / cosmetic keywords even if color check fails
					local frameStr = string.lower(tostring(v.Name))
					if frameStr:find("serum") or frameStr:find("halo") or frameStr:find("scar") then
						isSpecial = true
					end

					if isSpecial then
						local qty = inner:FindFirstChild("Quantity")
						local itemId = string.match(tostring(v.Name), "(%d+)$")
						local displayName = nil

						-- 1. Direct attribute check
						pcall(function()
							local attrName = v:GetAttribute("Name") or v:GetAttribute("ItemName") or v:GetAttribute("DisplayName")
							if attrName and tostring(attrName) ~= "" then
								displayName = CleanRewardName(attrName)
							end
						end)

						-- 2. Scrape every TextLabel inside the reward frame
						if not displayName then
							for _, desc in ipairs(v:GetDescendants()) do
								if desc:IsA("TextLabel") or desc:IsA("TextButton") then
									local txt = tostring(desc.Text or "")
									local lower = string.lower(txt)
									-- Accept real names, reject pure numbers / empty / EXP labels
									if txt ~= ""
										and not txt:match("^x?%d+$")
										and not txt:match("^%d+%s*%d+$")
										and lower ~= "exp"
										and lower ~= "xp"
										and #txt > 2
									then
										-- Prefer anything containing serum / scar / halo / known keywords
										if lower:find("serum") or lower:find("scar") or lower:find("halo")
											or lower:find("attack") or lower:find("armored")
											or lower:find("female") or lower:find("colossal")
											or lower:find("blazing") or lower:find("veil")
										then
											displayName = txt
											break
										end
										-- Keep as fallback candidate
										if not displayName then
											displayName = txt
										end
									end
								end
							end
						end

						-- 3. Hard-coded serum map from known IDs / frame names
						if not displayName or displayName:match("^[0-9%s]+$") then
							local serumMap = {
								["attack"] = "Attack Serum",
								["armored"] = "Armored Serum",
								["female"] = "Female Serum",
								["colossal"] = "Colossal Serum",
								["scar"] = "Female Scars",
								["blazing"] = "Blazing Veil",
							}
							local searchTarget = string.lower(tostring(v.Name) .. " " .. tostring(displayName or ""))
							for key, properName in pairs(serumMap) do
								if searchTarget:find(key) then
									displayName = properName
									break
								end
							end
						end

						-- 4. Cross-reference dropsData from server
						if (not displayName or displayName:match("^[0-9%s]+$")) and itemId and dropsData[itemId] then
							local dropName = CleanRewardName(dropsData[itemId])
							if dropName and not dropName:match("^[0-9%s]+$") and dropName ~= "Unknown" then
								displayName = dropName
							end
						end

						-- 5. Final cleanup + title case
						if displayName and not displayName:match("^[0-9%s]+$") then
							displayName = CleanRewardName(displayName)
							-- Title Case
							displayName = displayName:lower():gsub("(%a)([%w_']*)", function(first, rest)
								return first:upper() .. rest
							end)
						else
							-- Absolute last resort: check the whole frame tree one more time for serum strings
							local rawDump = ""
							pcall(function()
								for _, d in ipairs(v:GetDescendants()) do
									if d:IsA("TextLabel") then rawDump = rawDump .. " " .. tostring(d.Text) end
								end
							end)
							local lowerDump = string.lower(rawDump)
							if lowerDump:find("attack") and lowerDump:find("serum") then
								displayName = "Attack Serum"
							elseif lowerDump:find("armored") and lowerDump:find("serum") then
								displayName = "Armored Serum"
							elseif lowerDump:find("female") and lowerDump:find("serum") then
								displayName = "Female Serum"
							elseif lowerDump:find("colossal") and lowerDump:find("serum") then
								displayName = "Colossal Serum"
							elseif lowerDump:find("female") and lowerDump:find("scar") then
								displayName = "Female Scars"
							else
								displayName = "Mythical Drop"
							end
						end

						if not data.Special[displayName] then
							data.Special[displayName] = qty and tostring(qty.Text) or "1"
						end
					end
				end
			end
		end
	end

	pcall(function()
		local currentSlot = player:GetAttribute("Slot") or "A"
		if mapData and mapData.Slots and mapData.Slots[currentSlot] then
			local sd = mapData.Slots[currentSlot]
			if sd.Currency then
				for i, v in pairs(sd.Currency) do
					if i == "Gems" or i == "Gold" then data.Total[i] = v end
				end
			end
			if sd.Progression then
				for i, v in pairs(sd.Progression) do
					if i == "Prestige" or i == "Level" or i == "Streak" then data.Total[i] = v end
				end
			end
		end
	end)

	return data
end

		local missionElapsedSecs = 0
		if getgenv().KiraHub.State.MissionStartTime then
			missionElapsedSecs = os.clock() - getgenv().KiraHub.State.MissionStartTime
		end

		getgenv().KiraHub.State._missionStartClock = nil
		getgenv().KiraHub.State.MissionStartTime = os.clock()
		getgenv().KiraHub.State.GamesPlayed = getgenv().KiraHub.State.GamesPlayed + 1
		pcall(function() writefile(CFG_GAMES, tostring(getgenv().KiraHub.State.GamesPlayed)) end)
		
		pcall(function()
			local res = getRemote:InvokeServer("S_Rewards", "Get", true)
			if res and res.Obtained then
				getgenv().KiraHub.State.TotalGold = getgenv().KiraHub.State.TotalGold + (res.Obtained.Gold or 0)
				getgenv().KiraHub.State.TotalGems = getgenv().KiraHub.State.TotalGems + (res.Obtained.Gems or 0)
				getgenv().KiraHub.State.TotalXP = getgenv().KiraHub.State.TotalXP + (res.Obtained.XP or 0)
			end
		end)

		local waitStart = os.clock()
		repeat
			task.wait(0.1)
			local hasData = false
			if statsFrame then
				for _, v in ipairs(statsFrame:GetChildren()) do
					if v:IsA("Frame") and v:FindFirstChild("Amount")
						and v.Amount.Text ~= "0" and v.Amount.Text ~= ""
					then hasData = true break end
				end
			end
			if hasData then break end
		until os.clock() - waitStart > 2

		local rewardsData = ScrapeMissionRewards()
		local personalDrops = {}
		local overallHighestRarity = "Legendary"

		if rewardsData.Special and next(rewardsData.Special) then
			getgenv().KiraHub.State.MythicalDrops = getgenv().KiraHub.State.MythicalDrops + 1
			for mythicalName, mythicalQty in pairs(rewardsData.Special) do
				local cName = CleanRewardName(mythicalName) or tostring(mythicalName)
				local lowerName = string.lower(cName)
				local rarityClass = "Legendary"

				if lowerName:find("serum") or lowerName:find("halo") or lowerName:find("scar")
					or lowerName:find("skin") or lowerName:find("prestige")
					or GetPerkRarity(mythicalName) == "Secret"
				then
					rarityClass = "Mythical"
					overallHighestRarity = "Mythical"
				elseif GetPerkRarity(mythicalName) == "Legendary" then
					rarityClass = "Legendary"
				end

				table.insert(personalDrops, {
					Name = cName,
					Qty = tostring(mythicalQty),
					Rarity = rarityClass
				})

				DispatchMythicalWebhook(mythicalName, mythicalQty)
			end
		end

		if #personalDrops > 0 then
			local missionInfo = {
				Map = (mapData and mapData.Map and mapData.Map.Name) or "Unknown",
				Objective = (mapData and mapData.Map and mapData.Map.Objective) or "Unknown",
				Difficulty = (mapData and mapData.Map and mapData.Map.Difficulty) or "Unknown",
				Type = (mapData and mapData.Map and mapData.Map.Type) or "Missions"
			}
			SendPersonalMythicalWebhook(missionInfo, overallHighestRarity, personalDrops, missionElapsedSecs)
		end

		SaveSession()
		
		local returnCounter = tonumber(readfile(CFG_RETURN_COUNTER)) or 0
		if IsToggled("AutoReturnLobbyToggle") then
			returnCounter = returnCounter + 1
			if returnCounter >= GetOption("ReturnAfterGamesSlider", 10) then
				writefile(CFG_RETURN_COUNTER, "0")
				task.spawn(function() getRemote:InvokeServer("Functions", "Teleport", "Lobby") end)
				task.wait(0.5)
				TeleportService:Teleport(LOBBY_ID, player)
				return
			end
			writefile(CFG_RETURN_COUNTER, tostring(returnCounter))
		end

		if IsToggled("ToggleRewardWebhook") then
			DispatchMissionWebhook(rewardsData)
		end
	end)
end

local function GetTSSlotData()
	local slot = player:GetAttribute("Slot")
	local ok, result = pcall(function() return getRemote:InvokeServer("Functions", "Settings", "Blur", "Off") end)
	if ok and type(result) == "table" and result.Slots then
		return slot and result.Slots[slot] or result.Slots.A
	end
	return nil
end

local function GetTSQuestTable()
	local data = GetTSSlotData()
	if data and type(data.Quests) == "table" then
		return data.Quests.Spears
	end
	return nil
end

local function GetTSQuestProgress(tag)
	local questTable = GetTSQuestTable()
	if type(questTable) ~= "table" then return 0, false, false end
	for _, q in pairs(questTable) do
		if type(q) == "table" and tag == q.Tag then
			return q.Current or 0, q.Rewarded == true, true
		end
	end
	return 0, false, false
end

local function GetTSInventoryPieceCount()
	local data = GetTSSlotData()
	local inv = data and data.Inventory
	if type(inv) ~= "table" then return 0 end
	local count = 0
	for _, piece in ipairs(getgenv().KiraHub.Data.TSPieces) do
		local owned = false
		local items = inv.Items
		if items and type(items) == "table" and type(items[piece]) == "number" and items[piece] >= 1 then
			owned = true
		end
		if not owned then
			for _, category in pairs(inv) do
				if type(category) == "table" and type(category[piece]) == "number" and category[piece] >= 1 then
					owned = true
					break
				end
			end
		end
		if owned then count = count + 1 end
	end
	return count
end

local function IsTSFullyComplete()
	if GetTSInventoryPieceCount() >= 3 then return true end
	local completed = 0
	for _, stage in ipairs(getgenv().KiraHub.Data.TSStages) do
		if stage.Piece then
			local _, rewarded = GetTSQuestProgress(stage.Tag)
			if rewarded then completed = completed + 1 end
		end
	end
	return completed >= 3
end

local function CheckTSPrereq(stage)
	if not stage.PreReq then return true end
	for _, s in ipairs(getgenv().KiraHub.Data.TSStages) do
		if s.Tag == stage.PreReq then
			local current, rewarded = GetTSQuestProgress(s.Tag)
			return rewarded or (current >= s.Amount)
		end
	end
	return true
end

local function FindNextTSStage()
	for _, stage in ipairs(getgenv().KiraHub.Data.TSStages) do
		local current, rewarded, active = GetTSQuestProgress(stage.Tag)
		if not rewarded and CheckTSPrereq(stage) then
			return stage, current, active
		end
	end
	return nil
end

local function ClaimTSQuestRewards()
	local claimed = 0
	for _, stage in ipairs(getgenv().KiraHub.Data.TSStages) do
		local current, rewarded, active = GetTSQuestProgress(stage.Tag)
		if active and not rewarded and current >= stage.Amount then
			local ok, result = pcall(getRemote.InvokeServer, getRemote, "Functions", "Quest", stage.Tag, "Spears")
			if ok and result ~= nil then
				claimed = claimed + 1
			end
			task.wait(0.2)
		end
	end
	return claimed
end

local function CollectTSSupplies()
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return 0 end
	local crates = {}
	for _, v in ipairs(workspace:GetDescendants()) do
		if (v:IsA("Model") or v:IsA("BasePart")) and string.find(v.Name, "ThunderSpear_Supplies", 1, true) then
			table.insert(crates, v)
		end
	end
	if #crates == 0 then return 0 end
	local waitTime = GetOption("TSPickupWaitSlider", 4)
	local delivered = 0
	for _, crate in ipairs(crates) do
		if not IsToggled("AutoTSQuestToggle") then break end
		local pivot
		if crate:IsA("Model") then
			pivot = crate:GetPivot()
		else
			pivot = crate.CFrame
		end
		root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
		root.CFrame = pivot * CFrame.new(0, 3, 3)
		task.wait(waitTime)
		local circle = GetSuppliesCircle()
		if circle then
			local circlePivot
			if circle:IsA("Model") then
				circlePivot = circle:GetPivot()
			else
				circlePivot = circle.CFrame
			end
			root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			root.CFrame = circlePivot * CFrame.new(0, 3, 0)
			task.wait(2)
			delivered = delivered + 1
		end
	end
	return delivered
end

local function ExecuteTSQuestLoop()
	if not IsToggled("AutoTSQuestToggle") then
		pcall(function() getgenv().KiraHub.UI.Labels.TSStatus:SetText("Status: Idle") end)
		return
	end

	pcall(ClaimTSQuestRewards)

	if IsTSFullyComplete() then
		pcall(function() getgenv().KiraHub.UI.Labels.TSStatus:SetText("Status: All 3 Pieces Acquired!") end)
		return
	end

	local prestige = player:GetAttribute("Prestige")
	if not prestige then
		local data = GetTSSlotData()
		prestige = data and data.Progression and data.Progression.Prestige or 0
	end

	local minPrestige = GetOption("ThunderSpearPrestigeSlider", 4)
	if prestige < minPrestige then
		pcall(function() getgenv().KiraHub.UI.Labels.TSStatus:SetText("Waiting for Prestige " .. minPrestige) end)
		return
	end

	local stage, current, active = FindNextTSStage()
	if not stage then
		pcall(function() getgenv().KiraHub.UI.Labels.TSStatus:SetText("Status: Waiting for stage claim") end)
		return
	end

	pcall(function() getgenv().KiraHub.UI.Labels.TSStatus:SetText("TS: " .. stage.Tag .. " (" .. current .. "/" .. stage.Amount .. ")") end)

	if isLobby then
		local myMission = nil
		for _, m in ipairs(ReplicatedStorage.Missions:GetChildren()) do
			if m:FindFirstChild("Leader") and m.Leader.Value == player.Name then
				myMission = m
				break
			end
		end
		if myMission then
			if myMission.Name ~= stage.Map then
				pcall(function() getRemote:InvokeServer("S_Missions", "Leave") end)
				task.wait(1)
			else
				if stage.AllMods then
					for _, mod in ipairs(getgenv().KiraHub.Data.AllModifiers) do
						pcall(function() getRemote:InvokeServer("S_Missions", "Modify", mod) end)
					end
					task.wait(0.3)
				end
				pcall(function() getRemote:InvokeServer("S_Missions", "Start") end)
				task.wait(5)
			end
		else
			pcall(function()
				getRemote:InvokeServer("S_Missions", "Create", {
					Difficulty = stage.Difficulty or "Aberrant",
					Type = "Missions",
					Name = stage.Map,
					Objective = stage.Objective
				})
			end)
			task.wait(0.6)
		end
	else
		if stage.ForceRemote then
			pcall(function()
				getRemote:InvokeServer(stage.ForceRemote[1], stage.ForceRemote[2], stage.ForceRemote[3])
			end)
		end
		if stage.CollectCrates then
			pcall(CollectTSSupplies)
		end
	end
end

task.spawn(function()
	while true do
		task.wait(3)
		if not getgenv().KiraHubAOTR then break end
		pcall(ExecuteTSQuestLoop)
	end
end)

local function DetectBossSkin(bossModel, cleanSkinName)
	if not bossModel then return false, nil end
	local skinAttr = bossModel:GetAttribute("Skin")
	local fake = bossModel:FindFirstChild("Fake")
	local hair = fake and fake:FindFirstChild("Hair")
	local head = fake and fake:FindFirstChild("Head")
	local detectedSkin = skinAttr or (hair and hair.Name) or (head and head.Name) or "Standard"
	local matched = string.find(string.lower(tostring(detectedSkin)), string.lower(cleanSkinName)) ~= nil
	return matched, detectedSkin
end

local function CreateRaidForSkinHunter(raidMapName)
	local activeMissions = ReplicatedStorage:FindFirstChild("Missions")
	if not activeMissions then return false end
	
	local registeredMission = nil
	for _, m in ipairs(activeMissions:GetChildren()) do
		if m:FindFirstChild("Leader") and m.Leader.Value == player.Name then
			registeredMission = m
			break
		end
	end
	
	if registeredMission then
		if registeredMission.Name == raidMapName then
			pcall(function() getRemote:InvokeServer("S_Missions", "Start") end)
			return true
		else
			pcall(function() getRemote:InvokeServer("S_Missions", "Leave") end)
			task.wait(1)
		end
	end
	
	local difficulties = {"Aberrant", "Severe", "Hard"}
	local raidObjective = getgenv().KiraHub.Data.RaidObjectives[raidMapName] or "Attack Titan"
	
	for _, diff in ipairs(difficulties) do
		if not IsToggled("SkinHunterToggle") then return false end
		pcall(function()
			getRemote:InvokeServer("S_Missions", "Create", {
				Difficulty = diff,
				Type = "Raids",
				Name = raidMapName,
				Objective = raidObjective
			})
		end)
		task.wait(1)
		
		local created = false
		for _, m in ipairs(activeMissions:GetChildren()) do
			if m:FindFirstChild("Leader") and m.Leader.Value == player.Name then
				created = true
				break
			end
		end
		
		if created then
			task.wait(0.5)
			pcall(function() getRemote:InvokeServer("S_Missions", "Start") end)
			return true
		end
	end
	
	return false
end

local function ForceRetryToRejoin()
	local rewardsGui = INTERFACE:FindFirstChild("Rewards")
	if rewardsGui and rewardsGui.Visible then
		return ForceRetry()
	end
	pcall(function() getRemote:InvokeServer("Functions", "Retry", "Add") end)
	task.wait(0.3)
	pcall(function() postRemote:FireServer("Functions", "Retry") end)
	return true
end

local function ExecuteSkinHunterLoop()
	if not IsToggled("SkinHunterToggle") then
		pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: Idle") end)
		return
	end

	local targetSkin = GetOption("TargetSkinDropdown", "Onikiri Skin")
	local cleanSkinName = string.gsub(targetSkin, " Skin", "")
	local targetRaidMap = getgenv().KiraHub.Data.SkinToRaidMap[cleanSkinName]
	local bossName = getgenv().KiraHub.Data.SkinToBoss[cleanSkinName]
	
	if not targetRaidMap or not bossName then return end

	if isLobby then
		pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: Queueing " .. targetRaidMap) end)
		CreateRaidForSkinHunter(targetRaidMap)
	elseif isMenu then
		pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: In Main Menu") end)
		task.wait(3)
	else
		pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: Scanning Boss...") end)
		local titans = workspace:FindFirstChild("Titans")
		if not titans then
			task.wait(3)
			return
		end
		
		local waitStart = os.clock()
		local bossFound = nil
		while os.clock() - waitStart < 15 and IsToggled("SkinHunterToggle") do
			for _, t in ipairs(titans:GetChildren()) do
				if t.Name == bossName then
					bossFound = t
					break
				end
			end
			if bossFound then break end
			task.wait(1)
		end
		
		if not IsToggled("SkinHunterToggle") then return end
		
		if bossFound then
			local matched, detectedSkin = DetectBossSkin(bossFound, cleanSkinName)
			if matched then
				pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: TARGET FOUND!") end)
				SetToggle("SkinHunterToggle", false)
				
				if not getgenv().KiraHub.SkinAlreadyLogged[cleanSkinName] then
					getgenv().KiraHub.SkinAlreadyLogged[cleanSkinName] = true
					getgenv().KiraHub.DispatchSkinHunter(cleanSkinName, bossFound.Name)
				end
				
				Library:Notify({
					Title = "Skin Hunter Match!",
					Description = cleanSkinName .. " found on " .. bossFound.Name,
					Time = 10
				})
				return
			else
				pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: Wrong Skin - Rejoining") end)
				local delayTime = GetOption("RejoinSpeedSlider", 10)
				task.wait(delayTime)
				if IsToggled("SkinHunterToggle") then
					ForceRetryToRejoin()
					task.wait(3)
					if not workspace:FindFirstChild("Titans") then
						pcall(function() getRemote:InvokeServer("Functions", "Teleport", "Lobby") end)
						task.wait(0.5)
						pcall(function() TeleportService:Teleport(LOBBY_ID, player) end)
					end
				end
			end
		else
			pcall(function() getgenv().KiraHub.UI.Labels.SkinHunter:SetText("Status: No Boss - Rejoining") end)
			local delayTime = GetOption("RejoinSpeedSlider", 10)
			task.wait(delayTime)
			if IsToggled("SkinHunterToggle") then
				ForceRetryToRejoin()
				task.wait(3)
				if not workspace:FindFirstChild("Titans") then
					pcall(function() getRemote:InvokeServer("Functions", "Teleport", "Lobby") end)
					task.wait(0.5)
					pcall(function() TeleportService:Teleport(LOBBY_ID, player) end)
				end
			end
		end
	end
end

task.spawn(function()
	while true do
		task.wait(5)
		if not getgenv().KiraHubAOTR then break end
		pcall(ExecuteSkinHunterLoop)
	end
end)

local function IdentifyFamilyRarity(clanTitleText)
	local rarity = string.lower(string.match(clanTitleText, "%((.-)%)") or "")
	return rarity
end

local function ExecuteFamilyRoll(targets, rarities)
	if not INTERFACE:FindFirstChild("Customisation") then return end
	if not INTERFACE.Customisation.Visible then return end
	
	local familyString = ""
	pcall(function() familyString = INTERFACE.Customisation.Family.Family.Title.Text end)
	if familyString == "" then return end
	
	pcall(function() getgenv().KiraHub.UI.Labels.Family:SetText("Family: " .. familyString) end)
	
	local familyName = targets and string.lower(string.split(familyString, " ")[1]) or nil
	local familyRarity = IdentifyFamilyRarity(familyString)
	local stopRolling = false
	
	if targets and familyName and table.find(targets, familyName) then stopRolling = true end
	if rarities and table.find(rarities, familyRarity) then stopRolling = true end
	if familyRarity == "mythical" or familyRarity == "secret" then stopRolling = true end
	
	if stopRolling then
		
		if (familyRarity == "mythical" or familyRarity == "secret") and IsToggled("ToggleMythicalFamilyWebhook") then
			local url = getgenv().KiraHub.Webhooks.GetRewards()
			if url ~= "" then
				local rareMythicals = {"helos", "fritz", "reiss", "tybur"}
				local isRareMythical = false
				for _, name in ipairs(rareMythicals) do
					if string.find(string.lower(familyString), name) then isRareMythical = true break end
				end
				local liveBan = getgenv().KiraHub.State.BanStatus or "Unknown"
				
				getgenv().KiraHub.Webhooks.Post(url, {
					content = isRareMythical and "@everyone 👑 RARE MYTHICAL FAMILY!" or "💎 MYTHICAL FAMILY!",
					embeds = {{
						title = "👑 Kira Hub Premium | Family Roll",
						color = isRareMythical and 16711680 or 16750848,
						fields = {
							{name = "👤 Player", value = player.Name, inline = true},
							{name = "👥 Family", value = tostring(familyString), inline = true},
							{name = "💎 Rare Tier", value = isRareMythical and "YES" or "No", inline = true},
							{name = "🛡️ Ban Status", value = liveBan, inline = false}
						},
						footer = {text = "Kira Hub Premium v2.9"},
						timestamp = DateTime.now():ToIsoDate()
					}}
				})
			end
		end
		
		-- Smooth Hub "Store & keep spinning": store the keeper, and only count it as stored if the
		-- shown family actually changes within 4s (it doesn't when storage is full)
		if IsToggled("AutoStoreFamilyToggle") then
			local done = InvokeTimed(8, "Family", "Store")
			local changed = false
			if done then
				local t0 = os.clock()
				while os.clock() - t0 < 4 do
					task.wait(0.4)
					local nowFamily = ""
					pcall(function() nowFamily = INTERFACE.Customisation.Family.Family.Title.Text end)
					if nowFamily ~= familyString then changed = true break end
				end
			end
			if changed then
				Library:Notify({Title = "👑 Family Roller", Description = "Stored " .. familyString .. " - rolling on", Time = 4})
				return
			end
			SetToggle("AutoRollToggle", false)
			Library:Notify({Title = "👑 Family Roller", Description = "Could not store " .. familyString .. " (storage full?) - stopped", Time = 6})
			return
		end

		SetToggle("AutoRollToggle", false)
		Library:Notify({Title = "👑 Family Roller", Description = "Target rolled: " .. familyString, Time = 5})
		return
	end
	
	pcall(function()
		if INTERFACE:FindFirstChild("Warning") and INTERFACE.Warning.Prompt.Visible then
			UseButton(INTERFACE.Warning.Prompt.Main.Yes)
			task.wait(0.5)
		end
	end)
	
	pcall(function()
		local customisation = INTERFACE:FindFirstChild("Customisation")
		local famFrame = customisation and customisation:FindFirstChild("Family")
		if famFrame and not famFrame.Visible then
			UseButton(INTERFACE.Customisation.Categories.Family.Interact)
			task.wait(1)
		end
	end)
	
	pcall(function()
		local customisation = INTERFACE:FindFirstChild("Customisation")
		local famFrame = customisation and customisation:FindFirstChild("Family")
		local rollBtn = famFrame and famFrame:FindFirstChild("Buttons_2") and famFrame.Buttons_2:FindFirstChild("Roll")
		if rollBtn then UseButton(rollBtn) end
	end)
end

local function ClaimQuestCategory(profile, category)
	local list = profile.Quests and profile.Quests[category]
	if type(list) ~= "table" then return 0 end
	
	local claimed = 0
	for _, q in pairs(list) do
		if type(q) == "table" and q.Tag and q.Rewarded == nil then
			local maxVal = q.Amount or 0
			local curVal = tonumber(q.Current) or 0
			if maxVal > 0 and curVal >= maxVal then
				local ok, result = pcall(function()
					return getRemote:InvokeServer("Functions", "Quest", q.Tag, category)
				end)
				if ok and result ~= nil then
					claimed = claimed + 1
				end

local function ExecuteAutoEquipMetaPerks()
	if not IsToggled("AutoEquipMetaPerksToggle") then return end
	if not isLobby then return end
	
	local profile = GetPlayerData()
	local slot = player:GetAttribute("Slot")
	if not slot or not profile or not profile.Slots or not profile.Slots[slot] then return end
	
	local slotConfig = profile.Slots[slot]
	local storage = slotConfig.Perks and slotConfig.Perks.Storage or {}
	local equipped = slotConfig.Perks and slotConfig.Perks.Equipped or {}
	
	local offenseSet = false
	local bodySet = false
	local activeOffense = equipped.Offense
	local activeBody = equipped.Body
	
	if activeOffense and activeOffense ~= "" and storage[activeOffense] then
		local name = storage[activeOffense].Name
		if name == "Carnifex" or name == "Kengo" or name == "Butcher" then offenseSet = true end
	end
	if activeBody and activeBody ~= "" and storage[activeBody] then
		local name = storage[activeBody].Name
		if name == "Carnifex" or name == "Kengo" or name == "Butcher" then bodySet = true end
	end
	
	if offenseSet and bodySet then return end
	
	local metaCache = {}
	for id, tbl in pairs(storage) do
		if tbl.Name == "Carnifex" or tbl.Name == "Kengo" or tbl.Name == "Butcher" then
			local prio = tbl.Name == "Carnifex" and 1 or (tbl.Name == "Kengo" and 2 or 3)
			table.insert(metaCache, {id = id, name = tbl.Name, prio = prio})
		end
	end
	table.sort(metaCache, function(a, b) return a.prio < b.prio end)
	
	local usedIds = {}
	local changed = false
	
	if not offenseSet then
		for _, item in ipairs(metaCache) do
			if not usedIds[item.id] then
				local ok, success = pcall(function()
					return getRemote:InvokeServer("S_Equipment", "Perk_State", item.id, "Equip", "Offense")
				end)
				if ok and success then
					usedIds[item.id] = true
					changed = true
					Library:Notify({Title = "Auto Equip", Description = item.name .. " → Offense", Time = 3})
					break
				end
			end
		end
		task.wait(0.5)
	end
	
	if not bodySet then
		for _, item in ipairs(metaCache) do
			if not usedIds[item.id] then
				local ok, success = pcall(function()
					return getRemote:InvokeServer("S_Equipment", "Perk_State", item.id, "Equip", "Body")
				end)
				if ok and success then
					usedIds[item.id] = true
					changed = true
					Library:Notify({Title = "Auto Equip", Description = item.name .. " → Body", Time = 3})
					break
				end
			end
		end
	end
	
	if changed then
		pcall(function()
			local data = getRemote:InvokeServer("Data", "Copy")
			if data then mapData = data end
		end)
	end
end

	local targetWave = GetOption("WaveDieAtSlider", 20)
	local waveLabel = nil
	
	pcall(function()
		local hud = INTERFACE:FindFirstChild("HUD")
		if hud then
			for _, d in ipairs(hud:GetDescendants()) do
				if d:IsA("TextLabel") and (d.Name == "WaveValue" or d.Text:find("WAVE") or d.Text:find("Wave")) then
					waveLabel = d
					break
				end
			end
		end
	end)
	
	if waveLabel then
		local currentWave = tonumber(waveLabel.Text:match("(%d+)")) or 1
		if currentWave >= targetWave then
			local char = player.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum then hum.Health = 0 end
		end
	end
end

-- Each wave handler runs on its own loop, so one slow handler (crate run, a long upgrade pass)
-- can't hold up base upgrades, blessing or the shop.
do
	local function WavesLoop(interval, handler, anyMode)
		task.spawn(function()
			while true do
				task.wait(interval)
				if not getgenv().KiraHubAOTR then break end
				if anyMode or IsWavesMode() then pcall(handler) end
			end
		end)
	end
	WavesLoop(3, HandleWavesTSBuy)
	WavesLoop(3, HandleWavesBaseUpgrades)
	WavesLoop(3, HandleWaveDieAt)
	WavesLoop(3, HandleAutoWavesUpgrade)
	WavesLoop(3, HandleAutoCollectCrate)
	WavesLoop(3, HandleAutoStartWaves, true)
end

-- Smooth Hub ports

;(function()
	local KH = getgenv().KiraHub

	local UIS = game:GetService("UserInputService")

			pcall(function() l:SetText(text) end)
		end
	end

	----------------------------------------------------------------------------
	-- Shared GUI helpers (same click/visibility logic Smooth Hub uses)
	----------------------------------------------------------------------------
	local Gui = {}

		local t = tostring(obj.Text or ""):gsub("<[^>]->", "")

	end

	end
	function Gui.Visible(obj)

		local sg = obj:FindFirstAncestorOfClass("ScreenGui")
		return (sg and sg.IgnoreGuiInset) and Vector2.zero or GuiService:GetGuiInset()
	end
	local function mouseClick(p, withMove)
		return (pcall(function()
			if withMove then
				VirtualInputManager:SendMouseMoveEvent(p.X, p.Y, game)
				task.wait(0.05)
			end
			VirtualInputManager:SendMouseButtonEvent(p.X, p.Y, 0, true, game, 0)
			task.wait(withMove and 0.06 or 0.05)
			VirtualInputManager:SendMouseButtonEvent(p.X, p.Y, 0, false, game, 0)
		end))
	end
	local function selectEnter(btn, clearAfter)
		pcall(function() GuiService.SelectedObject = btn end)
		task.wait(0.1)
		local ok = pcall(function()
			VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
			task.wait(0.05)
			VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
		end)
		if clearAfter then
			task.wait(0.1)
			pcall(function() GuiService.SelectedObject = nil end)
		end
		return ok
	end

	-- prestige-style: four ways to press a button, tried in turn
	Gui.ClickWays = {
		function(b) return type(firesignal) == "function" and (pcall(firesignal, b.Activated)) end,
		function(b) return type(firesignal) == "function" and (pcall(firesignal, b.MouseButton1Click)) end,
		function(b) return selectEnter(b, false) end,
		function(b) return mouseClick(Gui.Center(b) + inset(b), false) end,
	}
	function Gui.ClickAny(b)
		for _, way in ipairs(Gui.ClickWays) do
			if way(b) then return true end
		end
		return false
	end

	-- shop/blessing-style: attempt n cycles firesignal -> real mouse click -> select + Enter
	function Gui.Press(b, attempt)
		attempt = attempt or 1
		local isButton = b:IsA("GuiButton")
		local way = (attempt - 1) % 3 + 1
		if way == 1 and (not isButton or type(firesignal) ~= "function") then way = 2 end
		if way == 1 then
			pcall(firesignal, b.Activated)
			pcall(firesignal, b.MouseButton1Click)
		elseif way == 2 then
			mouseClick(Gui.Center(b) + inset(b), true)
		elseif isButton then
			selectEnter(b, true)
		end
	end

			if p:IsA("GuiButton") then return p end

	function Gui.Root()

			if obj:IsA("BasePart") then return obj.Position end

			return part and part.Position
		end)

	-- NPC trips: tween at ~900 studs/s (0.15-6s), velocity held at zero, then snap to the goal
	function Gui.FlyTo(cf)
		local root = Gui.Root()
		if not root then return end
		pcall(function() player:RequestStreamAroundAsync(cf.Position, 0.5) end)
		local dur = math.clamp((root.Position - cf.Position).Magnitude / 900, 0.15, 6)
		local ok, tween = pcall(function()
			return TweenService:Create(root, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = cf})
		end)
		if not ok or not tween then
			pcall(function() root.CFrame = cf end)
			return
		end
		local hold = RunService.Heartbeat:Connect(function()
			pcall(function() root.AssemblyLinearVelocity = Vector3.zero end)
		end)
		tween:Play()
		local t0 = os.clock()
		while os.clock() - t0 < dur + 0.3 and tween.PlaybackState == Enum.PlaybackState.Playing and alive() do
			task.wait()
		end
		pcall(function() tween:Cancel() end)
		hold:Disconnect()
		pcall(function() root.CFrame = cf end)
	end

	-- "2,000" / "1.5k" / "3m" -> number
	function Gui.Amount(s)
		local num, suffix = tostring(s):match("^([%d,%.]+)%s*([kKmMbB]?)$")
		local n = num and tonumber((num:gsub(",", "")))
		if not n then return nil end
		suffix = suffix:lower()
		if suffix == "k" then n = n * 1e3 elseif suffix == "m" then n = n * 1e6 elseif suffix == "b" then n = n * 1e9 end
		return n

	end

	----------------------------------------------------------------------------
	-- Auto Skill: presses slot keys 1-5. Instant = tap, Hold = hold for Hold Time.
	-- On touch-only devices it taps the on-screen skill button instead.
	----------------------------------------------------------------------------
	do
		local KEYS = {Enum.KeyCode.One, Enum.KeyCode.Two, Enum.KeyCode.Three, Enum.KeyCode.Four, Enum.KeyCode.Five}
		local nextPress = {0, 0, 0, 0, 0}

		local function anySlotOn()
			for i = 1, 5 do
				if IsToggled("AutoSkill" .. i .. "Toggle") then return true end
			end
			return false
		end
		local function canUse()
			if not inMatch() then return false, "Not available in the lobby / menu" end
			if UIS:GetFocusedTextBox() then return false, "Typing in chat - paused" end
			local char = player.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not hum or hum.Health <= 0 then return false, "Waiting for your character" end
			local rewards = INTERFACE:FindFirstChild("Rewards")
			if rewards and rewards.Visible then return false, "Mission completed - paused" end
			return true

		local function skillButton(slot)
			for _, d in ipairs(pGui:GetDescendants()) do
				if d:IsA("GuiButton") and d.Visible and d.AbsoluteSize.X > 0 and d.AbsoluteSize.Y > 0 then
					local n = d.Name:lower()
					if (n:find("skill") or n:find("ability") or n:find("slot")) and n:find(tostring(slot)) then
						return d
					end
				end
			end
			return nil
		end
		local function tap(btn, hold)
			local p = Gui.Center(btn)
			local id = math.random(100000, 999999)
			pcall(function() VirtualInputManager:SendTouchEvent(id, 0, p.X, p.Y) end)
			task.wait(hold or 0.06)
			pcall(function() VirtualInputManager:SendTouchEvent(id, 2, p.X, p.Y) end)
		end
		local function useSlot(slot)
			local hold = GetOption("AutoSkill" .. slot .. "ModeDropdown", "Instant") == "Hold"
			local holdTime = GetOption("AutoSkillHoldTimeSlider", 3)
			if UIS.TouchEnabled and not UIS.MouseEnabled then
				local btn = skillButton(slot)
				if btn then tap(btn, hold and holdTime or nil) end
			elseif hold then
				pcall(function() VirtualInputManager:SendKeyEvent(true, KEYS[slot], false, game) end)
				local t0 = os.clock()
				while os.clock() - t0 < holdTime and alive() and IsToggled("AutoSkill" .. slot .. "Toggle") and canUse() do
					task.wait(0.1)
				end
				pcall(function() VirtualInputManager:SendKeyEvent(false, KEYS[slot], false, game) end)
			else
				pcall(function()
					VirtualInputManager:SendKeyEvent(true, KEYS[slot], false, game)
					task.wait(0.06)
					VirtualInputManager:SendKeyEvent(false, KEYS[slot], false, game)
				end)
			end
		end

		task.spawn(function()
			while alive() do
				task.wait(0.25)
				if not anySlotOn() then
					setLabel("SkillStatus", "Skill: Off")
				else
					local ok, why = canUse()
					if not ok then
						setLabel("SkillStatus", "Skill: " .. why)
					else
						setLabel("SkillStatus", "Skill: Waiting for next press...")
						for i = 1, 5 do
							if IsToggled("AutoSkill" .. i .. "Toggle") and os.clock() >= nextPress[i] then
								setLabel("SkillStatus", "Skill: Using slot " .. i .. " (" .. GetOption("AutoSkill" .. i .. "ModeDropdown", "Instant") .. ")")
								useSlot(i)
								nextPress[i] = os.clock() + GetOption("AutoSkillDelaySlider", 2)
								task.wait(0.15)
								if not canUse() then break end
							end
						end
					end
				end
			end
		end)

	----------------------------------------------------------------------------
	-- Auto Select Boost Mission: shows the boosted map + time left above the
	-- Mission Map dropdown and moves that map to the top of the list.
	----------------------------------------------------------------------------
	do
		local MAPS = {"Shiganshina", "Trost", "Outskirts", "Forest", "Utgard", "Docks", "Stohess", "Chapel"}
		local function scanBoost(inst)
			local ok, attrs = pcall(function() return inst:GetAttributes() end)
			if not ok or type(attrs) ~= "table" then return nil end
			local map, timeLeft
			for k, v in pairs(attrs) do
				local key = tostring(k):lower()
				if key:find("boost") then
					if type(v) == "string" and table.find(MAPS, v) then
						map = v
					elseif type(v) == "number" and (key:find("time") or key:find("left")) then
						timeLeft = v
					end
				end
			end
			return map, timeLeft
		end
		function KH.GetBoostedMission()
			local map, timeLeft = workspace:GetAttribute("Boosted_Map"), workspace:GetAttribute("Boosted_Timer")
			if type(map) ~= "string" or not table.find(MAPS, map) then
				map, timeLeft = nil, nil
				for _, inst in ipairs({workspace, ReplicatedStorage, player}) do
					local m, t = scanBoost(inst)
					if m then map, timeLeft = m, t break end
				end
			end
			if not map then return nil end
			if type(timeLeft) == "number" and timeLeft <= 3 then return nil end
			return map, (type(timeLeft) == "number") and timeLeft or nil
		end

		task.spawn(function()
			local lastText, lastTop
			while alive() do
				local map, timeLeft = KH.GetBoostedMission()
				local text = "Boosted Mission: none"
				if map then
					text = "Boosted Mission: " .. map
					if timeLeft then
						text = ("Boosted Mission: %s (%d:%02d left)"):format(map, math.floor(timeLeft / 60), math.floor(timeLeft % 60))
					end
				end
				if text ~= lastText then
					lastText = text
					setLabel("BoostMission", text)
				end
				if map ~= lastTop then
					lastTop = map
					local values = {}
					if map then values[1] = map end
					for _, m in ipairs(MAPS) do
						if m ~= map then values[#values + 1] = m end
					end
					pcall(function()
						local dd = Options.MissionMapDropdown
						local current = dd.Value
						dd:SetValues(values)
						if current then dd:SetValue(current) end
					end)
				end
				task.wait(1)
			end
		end)
	end

	----------------------------------------------------------------------------
	-- Auto Prestige (lobby only). Opens Equipment > Prestige, presses PRESTIGE,
	-- YES, picks the boost card for this prestige number, NEXT, then the memory
	-- card with the best priority (1 = first), then FINISH. Resumes mid-way if
	-- the screen is already on a later step.
	----------------------------------------------------------------------------
	do
		local BOOST_TEXT = {Luck = "luckboost", Gold = "goldboost", EXP = "expboost"}
		local busy, pending = false, false

		local function status(t) setLabel("PrestigeStatus", "Prestige: " .. tostring(t)) end
		local function gp(...)
			local o = pGui
			for _, n in ipairs({...}) do o = o and o:FindFirstChild(n) end
			return o
		end
		local function hasText(obj, want)
			if obj:IsA("TextButton") and Gui.Norm(obj.Text) == want then return true end
			for _, d in ipairs(obj:GetDescendants()) do
				if (d:IsA("TextLabel") or d:IsA("TextButton")) and Gui.Norm(d.Text) == want then return true end
			end
			return false
		end
		local function findButton(text, filter)
			local want, n = Gui.Norm(text), 0
			for _, d in ipairs(pGui:GetDescendants()) do
				n += 1
				if n % 1500 == 0 then task.wait() end
				if d:IsA("GuiButton") and Gui.Visible(d) and hasText(d, want) and (not filter or filter(d)) then
					return d
				end
			end
			return nil
		end
		local function findLabel(norm)
			local n = 0
			for _, d in ipairs(pGui:GetDescendants()) do
				n += 1
				if n % 1500 == 0 then task.wait() end
				if d:IsA("TextLabel") and Gui.Visible(d) and Gui.Norm(d.Text) == norm then return d end
			end
			return nil
		end
		local function nearText(obj, fragment)
			local p = obj.Parent
			for _ = 1, 5 do
				if not p or p == pGui then break end
				for _, d in ipairs(p:GetDescendants()) do
					if d:IsA("TextLabel") and Gui.Visible(d) and Gui.Norm(d.Text):find(fragment, 1, true) then return true end
				end
				p = p.Parent
			end
			return false
		end
		-- keep pressing getTarget() (rotating click ways) until done() or timeout
		local function clickUntil(getTarget, done, timeout)
			local stopAt, way = os.clock() + timeout, 1
			while os.clock() < stopAt and alive() and IsToggled("AutoPrestigeToggle") do
				if done() then return true end
				local target = getTarget()
				if target then
					Gui.ClickWays[way](target)
					local t0 = os.clock()
					while os.clock() - t0 < 1.5 do
						if done() then return true end
						task.wait(0.1)
					end
					way = way % #Gui.ClickWays + 1
				else
					task.wait(0.4)
				end
			end
			return done()
		end
		-- first visible button around a card label that doesn't belong to another card
		local function cardButton(label, others)
			local p = label.Parent
			for _ = 1, 6 do
				if not p or p == pGui then break end
				local list = p:GetDescendants()
				for _, d in ipairs(list) do
					if others[d] then return nil end
				end
				for _, d in ipairs(list) do
					if d:IsA("GuiButton") and Gui.Visible(d) then return d end
				end
				p = p.Parent
			end
			return nil
		end
		-- biggest frame around a card label (under 40% screen width) that holds no other card
		local function cardFrame(label, others)
			local best, p = nil, label.Parent
			local cam = workspace.CurrentCamera
			local maxW = cam and cam.ViewportSize.X * 0.4 or math.huge
			for _ = 1, 6 do
				if not p or p == pGui or not p:IsA("GuiObject") then break end
				if p.AbsoluteSize.X > maxW then break end
				local hit = false
				for _, d in ipairs(p:GetDescendants()) do
					if others[d] then hit = true break end
				end
				if hit then break end
				best = p
				p = p.Parent
			end
			return best
		end
		local function memoryCards()
			local byName = {}
			for _, m in ipairs(KH.Data.Memories) do
				byName[Gui.Norm((m.Name:gsub("^%[.-%]%s*", "")))] = m
			end
			local cards, n = {}, 0
			for _, d in ipairs(pGui:GetDescendants()) do
				n += 1
				if n % 1500 == 0 then task.wait() end
				if d:IsA("TextLabel") and Gui.Visible(d) then
					local mem = byName[Gui.Norm(d.Text)]
					if mem then
						local full = d:GetFullName()
						if not full:find(".Passives.", 1, true) and not full:find(".Storage.", 1, true) then
							cards[#cards + 1] = {label = d, mem = mem}
						end
					end
				end
			end
			return cards
		end
		local function currentPrestige() return tonumber(player:GetAttribute("Prestige")) or 0 end
		local function levelFull()
			local lvl = gp("Interface", "Equipment", "Prestige", "Progress", "Level")
			if not (lvl and lvl:IsA("TextLabel")) then return false, "Prestige page not found" end
			if Gui.Norm(lvl.Text):find("maxlvl", 1, true) then return true end
			local cur, max = lvl.Text:match("(%d+)%s*/%s*(%d+)")
			cur = tonumber(player:GetAttribute("Level")) or tonumber(cur)
			max = tonumber(max)
			if cur and max and cur >= max then return true end
			return false, ("Level not full ( %s / %s )"):format(tostring(cur), tostring(max))
		end
		local function finishShown() return findButton("FINISH") ~= nil end
		local function boostShown() return findLabel("luckboost") ~= nil end
		local function yesButton()
			return findButton("YES", function(b) return nearText(b, "areyousureyouwanttoprestige") end)
		end
		local function priorityOf(mem)
			return tonumber(GetOption("MemPrio_" .. mem.Id, "5")) or 5
		end

		local function runPrestige(step)
			busy = true
			local nextNumber = math.clamp(currentPrestige() + 1, 1, 5)
			local ok, failedAt = pcall(function()
				if step <= 1 then
					status("Opening the Prestige page...")
					local function pageTarget()
						local equip = gp("Interface", "Equipment")
						if not (equip and Gui.Visible(equip)) then
							return gp("Interface", "Topbar", "Main", "Categories", "Equipment", "Interact")
						end
						return gp("Interface", "Equipment", "Categories", "Prestige", "Main", "Interact")
					end
					local function prestigeButton()
						local b = gp("Interface", "Equipment", "Prestige", "B_Prestige")
						return (b and Gui.Visible(b)) and b or nil
					end
					if not clickUntil(pageTarget, function() return prestigeButton() ~= nil end, 12) then
						return "the Prestige page"
					end
					status("Clicking PRESTIGE...")
					if not clickUntil(prestigeButton, function() return yesButton() ~= nil end, 10) then
						return "the PRESTIGE button"
					end
				end
				if step <= 3 then
					status("Confirming with YES...")
					if not clickUntil(yesButton, boostShown, 10) then return "the YES button" end
				end
				if step <= 4 then
					local boost = GetOption("PrestigeBoost" .. nextNumber .. "Dropdown", "Luck")
					status(("Choosing %s boost (prestige #%d)..."):format(boost, nextNumber))
					local card = BOOST_TEXT[boost] and findLabel(BOOST_TEXT[boost])
					if card then
						local others = {}
						for name, norm in pairs(BOOST_TEXT) do
							if name ~= boost then
								local l = findLabel(norm)
								if l then others[l] = true end
							end
						end
						for attempt = 1, 3 do
							local btn = cardButton(card, others)
							local frame = cardFrame(card, others)
							if attempt == 1 then
								if frame then Gui.ClickWays[4](frame) elseif btn then Gui.ClickAny(btn) end
							elseif attempt == 2 then
								if btn then Gui.ClickAny(btn) elseif frame then Gui.ClickWays[4](frame) end
							else
								Gui.ClickWays[4](card)
							end
							local picked, t0 = false, os.clock()
							while os.clock() - t0 < 1.5 do
								task.wait(0.3)
								if findButton("NEXT") then picked = true break end
							end
							if picked then break end
						end
					else
						status("Boost card not found: " .. tostring(boost))
					end
					status("Clicking NEXT...")
					if not clickUntil(function() return findButton("NEXT") end, finishShown, 10) then
						return "the NEXT button"
					end
				end
				task.wait(0.8)
				local cards = memoryCards()
				if #cards > 0 then
					-- 1 = most wanted; ties go to more stars, then the leftmost card
					table.sort(cards, function(a, b)
						local pa, pb = priorityOf(a.mem), priorityOf(b.mem)
						if pa ~= pb then return pa < pb end
						if a.mem.Stars ~= b.mem.Stars then return a.mem.Stars > b.mem.Stars end
						return a.label.AbsolutePosition.X < b.label.AbsolutePosition.X
					end)
					local best = cards[1]
					status("Picking card " .. best.mem.Name .. " ...")
					local others = {}
					for i = 2, #cards do others[cards[i].label] = true end
					local frame = cardFrame(best.label, others)
					if frame then
						Gui.ClickWays[4](frame)
						task.wait(0.6)
					else
						local btn = cardButton(best.label, others)
						if btn then Gui.ClickAny(btn) task.wait(0.6) end
					end
				else
					status("No Memory cards found - keeping the current pick")
				end
				status("Clicking FINISH...")
				if not clickUntil(function() return findButton("FINISH") end, function() return not finishShown() end, 12) then
					return "the FINISH button"
				end
				return nil
			end)
			busy = false
			if not ok then status("Error: " .. tostring(failedAt)) return false end
			if failedAt then status("Failed at: " .. failedAt .. " (retrying in 30s)") return false end
			status(("Prestige done (now Prestige %d)"):format(currentPrestige()))
			return true
		end

		task.spawn(function()
			local nextTry, ticks = 0, 0
			while alive() do
				task.wait(1)
				ticks += 1
				if not IsToggled("AutoPrestigeToggle") then
					pending = false
					if ticks % 5 == 0 then status("Off") end
				elseif not isLobby then
					pending = false
					status("Lobby place only")
				elseif not busy and os.clock() >= nextTry then
					local stopAt = GetOption("PrestigeStopAtSlider", 3)
					local p = currentPrestige()
					if p >= stopAt then
						pending = false
						status(("Stopped: Prestige %d reached the target %d"):format(p, stopAt))
					else
						local step
						local full, why = levelFull()
						if ticks % 5 == 0 and (full or pending or ticks <= 30) then
							if finishShown() then step = 5
							elseif boostShown() then step = 4
							elseif yesButton() then step = 3 end
						end
						if not step then
							if full then
								step = 1
							else
								pending = false
								status(why or "Waiting for max level")
							end
						end
						if step then
							pending = true
							local done = runPrestige(step)
							if done then pending = false end
							nextTry = os.clock() + (done and 10 or 30)
						end
					end
				end
			end
		end)
	end

	local function waveNpc(name)
		local waves = workspace:FindFirstChild("Waves")
		local npcs = waves and waves:FindFirstChild("NPCs")
		return npcs and npcs:FindFirstChild(name)

	-- walk up to an NPC, press E (or fire its prompt), click the dialogue option,
	-- up to 3 tries; isOpen() tells when the right window is showing
	local function openNpc(npcName, statusKey, prefix, dialogueMatch, isOpen)
		if isOpen() then return true end
		local npc = waveNpc(npcName)
		local pos = npc and Gui.PosOf(npc)
		if not pos then
			setLabel(statusKey, prefix .. npcName .. " not found (not in wave mode?)")
			return false
		end
		for attempt = 1, 3 do
			if not Gui.Root() then return false end
			setLabel(statusKey, prefix .. ("Going to %s (%d/3)"):format(npcName, attempt))
			Gui.FlyTo(CFrame.new(pos + Vector3.new(4, 2, 4), pos))
			task.wait(0.4)
			setLabel(statusKey, prefix .. "Talking to " .. npcName .. " (E)")
			local prompt = npc:FindFirstChildWhichIsA("ProximityPrompt", true)
			if prompt and type(fireproximityprompt) == "function" and attempt ~= 2 then
				pcall(fireproximityprompt, prompt)
			else
				Gui.PressKey(Enum.KeyCode.E, (prompt and prompt.HoldDuration or 0) + 0.2)
			end
			local t0, clicked = os.clock(), false
			while os.clock() - t0 < 4 do
				task.wait(0.25)
				if isOpen() then return true end
				local opt = Gui.FindText(dialogueMatch)
				if opt and not clicked then
					setLabel(statusKey, prefix .. "Clicking the dialogue option")
					Gui.Press(Gui.ButtonOf(opt), attempt)
					clicked = true
					t0 = os.clock() - 2
				end
			end
		end
		return false
	end

		local running = false
		local bought, spent = 0, 0
		local P = "Shop: "

		local function status(t) setLabel("ShopStatus", P .. tostring(t)) end
		local function multi(id, name)

		local function activeTab(root)

		-- the row of 40-160px buttons above the tab title, left to right
		local function tabButtons(root)
			local cam = workspace.CurrentCamera
			local view = cam and cam.ViewportSize or Vector2.new(1920, 1080)
			local _, title = activeTab(root)
			local limitY = title and title.AbsolutePosition.Y or view.Y * 0.25
			local found = {}
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("GuiButton") and Gui.Visible(d) then
					local s = d.AbsoluteSize
					if s.X >= 40 and s.X <= 160 and s.Y >= 40 and s.Y <= 160 and Gui.Center(d).Y < limitY then
						local dup = false
						for _, f in ipairs(found) do
							if math.abs(Gui.Center(f).X - Gui.Center(d).X) < 25 and math.abs(Gui.Center(f).Y - Gui.Center(d).Y) < 25 then
								dup = true
								break
							end
						end

					end
				end
			end
			local row = {}
			for _, a in ipairs(found) do
				local same = {}
				for _, b in ipairs(found) do
					if math.abs(Gui.Center(a).Y - Gui.Center(b).Y) < 30 then same[#same + 1] = b end
				end
				if #same > #row then row = same end
			end
			table.sort(row, function(a, b) return Gui.Center(a).X < Gui.Center(b).X end)
			return row
		end
		-- the selected tab is the one clearly wider than the rest
		local function widestIndex(row)
			local bestIdx, top, second = nil, 0, 0
			for i, b in ipairs(row) do
				local w = b.AbsoluteSize.X
				if w > top then second = top top = w bestIdx = i
				elseif w > second then second = w end
			end
			if bestIdx and top - second >= 6 then return bestIdx end
			return nil
		end
		local function onTab(root, idx)
			local t = activeTab(root)
			if t then return t == idx end
			local w = widestIndex(tabButtons(root))
			if w then return w == idx end
			return nil
		end
		local function openTab(idx)
			local tries = {{0, 1}, {0, 2}, {1, 2}, {-1, 2}, {1, 1}, {-1, 1}, {0, 3}}
			for _, try in ipairs(tries) do
				local root = marketGui()
				if not root then return false end
				if onTab(root, idx) == true then return true end
				local btn = tabButtons(root)[idx + try[1]]
				if btn then Gui.Press(btn, try[2]) end
				local t0 = os.clock()
				while os.clock() - t0 < 1.0 do
					task.wait(0.1)
					local r = marketGui()
					if r then
						local state = onTab(r, idx)
						if state == true or (state == nil and try[1] == 0 and btn) then
							task.wait(0.35)
							return true
						end
					end
				end
			end
			return false
		end
		local function stockOf(text)
			local a, b = text:lower():gsub("o", "0"):match("^(%d+)%s*/%s*(%d+)$")
			if a then return tonumber(a), tonumber(b) end
			return nil
		end
		local function plainNumber(text)
			local s = text:gsub("[,%.%s]", "")
			if s:match("^%d+$") then return tonumber(s) end
			return nil
		end
		local function starsOf(card, priceY)
			for _, d in ipairs(card:GetDescendants()) do
				if d:IsA("TextLabel") and Gui.Visible(d) then
					local n, text = 0, d.Text or ""
					for _, ch in ipairs(STAR_CHARS) do n += select(2, text:gsub(ch, "")) end
					if n > 0 then return n end
				end
			end
			-- no star characters: count small icons on one row between the card middle and the price
			local rows = {}
			local minY = card.AbsolutePosition.Y + card.AbsoluteSize.Y * 0.55
			for _, d in ipairs(card:GetDescendants()) do
				if (d:IsA("ImageLabel") or d:IsA("ImageButton")) and Gui.Visible(d) then
					local s, c = d.AbsoluteSize, Gui.Center(d)
					if s.X <= 40 and s.Y <= 40 and c.Y > minY and c.Y < priceY - 2 then
						local key = math.floor(c.Y / 6)
						rows[key] = (rows[key] or 0) + 1
					end
				end
			end
			local best = 0
			for _, n in pairs(rows) do if n > best then best = n end end
			return best > 0 and best or nil
		end
		-- every card on the open tab: name, stock, price, stars, sold, crate
		local function readCards(root)
			local cards, seen = {}, {}
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("TextLabel") and Gui.Visible(d) and stockOf(Gui.Clean(d)) then
					local p, card = d.Parent, nil
					for _ = 1, 6 do
						if not p or p == root or p == pGui then break end
						if p:IsA("GuiObject") and p.AbsoluteSize.Y > 150 and p.AbsoluteSize.X < 600 then
							local hasPrice = false
							for _, e in ipairs(p:GetDescendants()) do
								if e:IsA("TextLabel") and e ~= d and Gui.Visible(e) and plainNumber(Gui.Clean(e)) then
									hasPrice = true
									break
								end
							end
							if hasPrice then card = p break end
						end
						p = p.Parent
					end
					if card and not seen[card] then
						seen[card] = true
						local info = {Obj = card, Sold = false}
						info.Count, info.Max = stockOf(Gui.Clean(d))
						local nameLbl, nameY, priceLbl, priceY = nil, math.huge, nil, -math.huge
						for _, e in ipairs(card:GetDescendants()) do
							if e:IsA("TextLabel") and Gui.Visible(e) and e ~= d then
								local t = Gui.Clean(e)
								if t:upper() == "SOLD" then
									info.Sold = true
								elseif plainNumber(t) then
									if e.AbsolutePosition.Y > priceY then priceLbl, priceY = e, e.AbsolutePosition.Y end
								elseif t ~= "?" and t:match("%a%a") and e.AbsolutePosition.Y < nameY then
									nameLbl, nameY = e, e.AbsolutePosition.Y
								end
							end
						end
						info.Name = nameLbl and Gui.Clean(nameLbl):upper() or "?"
						info.Price = priceLbl and plainNumber(Gui.Clean(priceLbl)) or nil
						info.PriceObj = priceLbl
						info.Stars = priceLbl and starsOf(card, priceLbl.AbsolutePosition.Y) or nil
						info.IsCrate = info.Name:find("CRATE") ~= nil
						if info.Price then cards[#cards + 1] = info end
					end
				end
			end
			table.sort(cards, function(a, b)
				local ca, cb = Gui.Center(a.Obj), Gui.Center(b.Obj)
				if math.abs(ca.Y - cb.Y) > 30 then return ca.Y < cb.Y end
				return ca.X < cb.X
			end)
			return cards
		end
		-- your coins: the lowest plain number on screen that isn't inside a card
		local function readCoins(root, cards)
			local value, y = nil, -math.huge
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("TextLabel") and Gui.Visible(d) then
					local n = plainNumber(Gui.Clean(d))
					if n then
						local inCard = false
						for _, c in ipairs(cards) do
							if d:IsDescendantOf(c.Obj) then inCard = true break end
						end
						if not inCard and d.AbsolutePosition.Y > y then value, y = n, d.AbsolutePosition.Y end
					end
				end
			end
			return value
		end
		local function cardButtonOf(card)
			if card:IsA("GuiButton") then return card end
			local best, area = nil, 0
			for _, d in ipairs(card:GetDescendants()) do
				if d:IsA("GuiButton") and Gui.Visible(d) then
					local text = d:IsA("TextButton") and Gui.Clean(d) or ""
					local a = d.AbsoluteSize.X * d.AbsoluteSize.Y
					if text ~= "?" and a > card.AbsoluteSize.X * card.AbsoluteSize.Y * 0.35 and a > area then best, area = d, a end
				end
			end
			return best
		end
		local function confirmIfAsked(cards)
			local c = Gui.FindText(function(t) return CONFIRM[t] == true end)
			if not c then return false end
			for _, card in ipairs(cards) do
				if c:IsDescendantOf(card.Obj) then return false end
			end
			Gui.Press(Gui.ButtonOf(c), 1)
			return true
		end
		local function wanted(card, coins, tab)
			if card.Sold or not card.Count or card.Count <= 0 or not card.Price then return false end
			if card.IsCrate and not multi("ShopCategoriesDropdown", "Crates") then return false end
			if card.Stars and not card.IsCrate then
				local opt = STAR_OPTION[tab] or STAR_OPTION[1]
				local starName = math.min(card.Stars, 5) == 1 and "1 Star" or (math.min(card.Stars, 5) .. " Stars")
				if not multi(opt, starName) then return false end
			end
			local maxPrice, keep = numInput("ShopMaxPriceInput"), numInput("ShopReserveInput")
			if maxPrice > 0 and card.Price > maxPrice then return false end
			if coins and card.Price > coins - keep then return false end
			return true
		end
		local function sortByOrder(list)
			local order = GetOption("ShopPriorityDropdown", "Highest star first")
			table.sort(list, function(a, b)
				if order == "Cheapest first" then
					if a.Price ~= b.Price then return a.Price < b.Price end
				elseif order == "Most expensive first" then
					if a.Price ~= b.Price then return a.Price > b.Price end
				else
					local sa, sb = a.Stars or 0, b.Stars or 0
					if sa ~= sb then return sa > sb end
					if a.Price ~= b.Price then return a.Price < b.Price end
				end
				return false
			end)
		end
		-- click a card until its stock drops / it shows SOLD / coins go down
		local function buy(card, root)
			local before = readCards(root)
			local coinsBefore = readCoins(root, before)
			local countBefore = card.Count
			local cardBtn = cardButtonOf(card.Obj)
			local priceBtn = (card.PriceObj and card.PriceObj.Parent) and Gui.ButtonOf(card.PriceObj) or nil
			for attempt = 1, 4 do
				if not alive() then return false end
				Gui.Press(cardBtn or priceBtn or card.Obj, attempt)
				local t0 = os.clock()
				while os.clock() - t0 < 1.8 do
					task.wait(0.2)
					local r = marketGui()
					if not r then return false end
					local now = readCards(r)
					confirmIfAsked(now)
					local c0 = Gui.Center(card.Obj)
					for _, c in ipairs(now) do
						local c1 = Gui.Center(c.Obj)
						if c.Name == card.Name and math.abs(c1.X - c0.X) < 20 and math.abs(c1.Y - c0.Y) < 20 then
							if c.Sold or (c.Count and countBefore and c.Count < countBefore) then return true end
						end
					end
					local coinsNow = readCoins(r, now)
					if coinsNow and coinsBefore and coinsNow < coinsBefore then return true end
				end
			end
			return false
		end
		local function shopTab(idx, tabName)
			if not openTab(idx) then return false, 0 end
			local skip, count = {}, 0
			for _ = 1, 40 do
				local root = marketGui()
				if not root or not alive() then break end
				local cards = readCards(root)
				local coins = readCoins(root, cards)
				if #cards == 0 then
					status(tabName .. ": no cards read")
				end
				local list = {}
				for _, c in ipairs(cards) do
					local c0 = Gui.Center(c.Obj)
					c.Key = c.Name .. "@" .. math.floor(c0.X / 10) .. "," .. math.floor(c0.Y / 10)
					if not skip[c.Key] and wanted(c, coins, idx) then list[#list + 1] = c end
				end
				if #list == 0 then break end
				sortByOrder(list)
				local pick = list[1]
				status(("%s: buying %s (%s coins)"):format(tabName, pick.Name, tostring(pick.Price)))
				if buy(pick, root) then
					count += 1
					bought += 1
					spent += pick.Price
					task.wait(0.3)
				else
					skip[pick.Key] = true
				end
			end
			return true, count
		end
		local function closeMarket()
			for _ = 1, 3 do
				local root = marketGui()
				if not root then return end
				local cam = workspace.CurrentCamera
				local view = cam and cam.ViewportSize or Vector2.new(1920, 1080)
				local best, score = nil, math.huge
				for _, d in ipairs(root:GetDescendants()) do
					if d:IsA("GuiButton") and Gui.Visible(d) then
						local c = Gui.Center(d)
						local text = d:IsA("TextButton") and Gui.Clean(d):upper() or ""
						if text == "X" or (c.Y < 120 and c.X > view.X - 160 and d.AbsoluteSize.X < 120) then
							local s = (view.X - c.X) + c.Y
							if s < score then best, score = d, s end
						end
					end
				end
				if best then Gui.Press(best, 1) end
				task.wait(0.6)
				if not marketGui() then break end
			end
			local bye = Gui.FindText(function(t) return t == "au revoir" or t == "goodbye" or t == "bye" end)
			if bye then
				Gui.Press(Gui.ButtonOf(bye), 1)
				task.wait(0.3)
			end
		end

		local function runShop()
			if running then return false end
			Gui.NpcAcquire("Shop")
			running = true
			local root = Gui.Root()
			local homeCF = root and root.CFrame
			bought, spent = 0, 0
			local ok, err = pcall(function()
				if not openNpc("Orange", "ShopStatus", P, function(t) return t:match("^%[?e%]?%s*market$") ~= nil end, function() return marketGui() ~= nil end) then
					status("Could not open the Market")
					error("open failed", 0)
				end
				task.wait(0.5)
				local tabs = {}
				if multi("ShopCategoriesDropdown", "Perks") then tabs[#tabs + 1] = {1, "Perks"} end
				if multi("ShopCategoriesDropdown", "Artifacts") then tabs[#tabs + 1] = {2, "Artifacts"} end
				if multi("ShopCategoriesDropdown", "Memories") then tabs[#tabs + 1] = {3, "Memories"} end
				if #tabs == 0 then
					status("Nothing selected to buy")
					return
				end
				for _, t in ipairs(tabs) do
					if not shopTab(t[1], t[2]) then status("Could not open tab: " .. t[2]) end
				end
				status(("Done: bought %d item(s), spent %d coins"):format(bought, spent))
			end)
			if not ok and err ~= "open failed" then
				status("Error: " .. tostring(err))
			end
			pcall(closeMarket)
			if Gui.Root() and homeCF then Gui.FlyTo(homeCF) end
			running = false
			Gui.NpcRelease("Shop")
			return ok
		end

		local wavesLabel, lastScan, lastWave = nil, 0, nil
		local function wavesCleared()
			if not (wavesLabel and wavesLabel.Parent) then
				wavesLabel = nil
				if os.clock() - lastScan < 2 then return lastWave end
				lastScan = os.clock()
				for _, d in ipairs(pGui:GetDescendants()) do
					if (d:IsA("TextLabel") or d:IsA("TextButton")) and Gui.Clean(d):lower():match("^waves cleared%s*%[%d+%s*/%s*%d+%]") then
						wavesLabel = d
						break
					end
				end
			end
			if not wavesLabel then return lastWave end
			local n = tonumber(Gui.Clean(wavesLabel):match("%[(%d+)%s*/") or "")
			if n then lastWave = n end
			return n or lastWave
		end

		KH.Shop = {RunNow = function()
			if not inMatch() then status("Not in a match") return end
			runShop()
		end}

		task.spawn(function()
			local doneWave, fails, nextTry = -1, 0, 0
			while alive() do
				task.wait(1)
				if not IsToggled("AutoShopToggle") then
					if not running then status("Off") end
					doneWave, fails = -1, 0
				elseif not inMatch() or not waveNpc("Orange") then
					status("Waiting for wave mode")
				elseif not running and not KH.State.Refilling and os.clock() >= nextTry then
					local wave = wavesCleared() or 0
					local cooldown = wave ~= doneWave and Gui.NpcCooldown("Shop")
					if cooldown then
						status(("Wave %d: %s"):format(wave, cooldown))
					elseif wave ~= doneWave then
						status(("Wave %d: shopping..."):format(wave))
						if runShop() then
							doneWave, fails = wave, 0
						else
							fails += 1
							nextTry = os.clock() + 6
							if fails >= 3 then doneWave, fails = wave, 0 end
						end
					else
						status(("Wave %d done. Waiting for next wave"):format(wave))
					end
				end
			end
		end)
	end

	----------------------------------------------------------------------------
	-- Auto Roll Blessings [ Roll ]: flies to Roll, opens the Blessings machine and
	-- rolls (2,000 gold each, up to 40 per visit) while you can afford it.
	-- Optionally stops for good once 3 cards match and a reward shows.
	----------------------------------------------------------------------------
	do
		local COST, MAX_ROLLS, ROLL_WAIT = 2000, 40, 3.5
		local GOLD_WORDS = {"gold", "coin", "money", "currenc", "cash"}
		local running = false
		local goldLabel, rewardLabel = nil, nil
		local P = "Blessings: "

		local function status(t) setLabel("BlessingStatus", P .. tostring(t)) end
		local function nearGold(obj)
			local o = obj
			for _ = 1, 5 do
				if not o or o == pGui then break end
				local n = o.Name:lower()
				for _, w in ipairs(GOLD_WORDS) do
					if n:find(w, 1, true) then return true end
				end
				o = o.Parent
			end
			return false
		end
		local function readGold(window)
			if goldLabel and goldLabel.Parent and Gui.Visible(goldLabel) then
				local v = Gui.Amount(Gui.Clean(goldLabel))
				if v then return v end
			end
			goldLabel = nil
			local best, rank, value = nil, -1, nil
			for _, d in ipairs(pGui:GetDescendants()) do
				if d:IsA("TextLabel") and Gui.Visible(d) and nearGold(d) then
					local v = Gui.Amount(Gui.Clean(d))
					if v then
						local r = (d.Parent and d.Parent:FindFirstChildWhichIsA("ImageLabel", true)) and 2 or 1
						if r > rank then best, rank, value = d, r, v end
					end
				end
			end
			if best then
				goldLabel = best
				return value
			end
			if window then
				local y, v2 = -math.huge, nil
				for _, d in ipairs(window:GetDescendants()) do
					if d:IsA("TextLabel") and Gui.Visible(d) then
						local v = Gui.Amount(Gui.Clean(d))
						if v and d.AbsolutePosition.Y > y then y, v2 = d.AbsolutePosition.Y, v end
					end
				end
				return v2
			end
			return nil
		end
		local function rollButton()
			return Gui.FindText(function(t) return t:match("^roll%s*%(") ~= nil end)
		end
		local function rolling()
			return Gui.FindText(function(t) return t:match("^rolling") ~= nil end)
		end
		local function rollCost()
			local b = rollButton()
			local m = b and Gui.Clean(b):match("%(([%d,%.]+%s*[kKmM]?)%)")
			return m and Gui.Amount((m:gsub("%s", ""))) or nil
		end
		local function blessingsWindow()
			local title = Gui.FindText(function(t) return t == "blessings" end)
			if not title then return nil end
			local p = title.Parent
			for _ = 1, 8 do
				if not p or p == pGui then break end
				for _, d in ipairs(p:GetDescendants()) do
					if (d:IsA("TextLabel") or d:IsA("TextButton")) and Gui.Clean(d):lower():match("^roll") then return p end
				end
				p = p.Parent
			end
			return title:FindFirstAncestorOfClass("ScreenGui") or pGui
		end
		local function closeWindow()
			for attempt = 1, 5 do
				local win = blessingsWindow()
				if not win then break end
				if not win:IsA("GuiObject") then
					local title = Gui.FindText(function(t) return t == "blessings" end)
					local o = title
					while o and o ~= pGui do
						if o:IsA("GuiObject") and o.AbsoluteSize.X >= 250 and o.AbsoluteSize.Y >= 300 then win = o break end
						o = o.Parent
					end
				end
				local pos, size
				if win:IsA("GuiObject") then
					pos, size = win.AbsolutePosition, win.AbsoluteSize
				else
					pos = Vector2.zero
					size = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1920, 1080)
				end
				local best, score = nil, math.huge
				for _, d in ipairs(win:GetDescendants()) do
					if d:IsA("GuiButton") and Gui.Visible(d) then
						local c = Gui.Center(d)
						local text = d:IsA("TextButton") and Gui.Clean(d):upper() or ""
						local n = d.Name:lower()
						local named = n:find("close", 1, true) or n:find("exit", 1, true)
						local corner = c.X > pos.X + size.X - 100 and c.Y < pos.Y + 100 and d.AbsoluteSize.X < 120
						if named or text == "X" or corner then
							local s = (pos.X + size.X - c.X) + (c.Y - pos.Y)
							if named then s -= 10000 end
							if s < score then best, score = d, s end
						end
					end
				end
				if best then
					status("Closing the Blessings window (X)")
					Gui.Press(best, attempt)
				end
				task.wait(0.6)
				if not blessingsWindow() then break end
			end
		end
		local function rewardState()
			if rewardLabel and rewardLabel.Parent then
				local t = Gui.Clean(rewardLabel)
				if t:lower():match("^no reward") then return "none" end
				if t ~= "" and Gui.Visible(rewardLabel) then return "reward", t end
				return nil
			end
			rewardLabel = nil
			local l = Gui.FindText(function(t) return t:match("^no reward") ~= nil end)
			if l then
				rewardLabel = l
				return "none"
			end
			return nil
		end
		local function rollLoop()
			local rolls, misses = 0, 0
			rewardLabel = nil
			for _ = 1, MAX_ROLLS do
				if not (alive() and IsToggled("AutoRollBlessingToggle")) then return rolls, "off" end
				local win = blessingsWindow()
				if not win then return rolls, "closed" end
				local gold = readGold(win)
				if gold == nil then return rolls, "nogold" end
				local cost = math.max(rollCost() or 0, COST)
				if gold < cost then return rolls, "poor" end
				local btn = rollButton()
				if not btn then
					local waited = 0
					while waited < 10 and alive() and IsToggled("AutoRollBlessingToggle") and blessingsWindow() and not rollButton() do
						task.wait(0.5)
						waited += 0.5
					end
					btn = rollButton()
					if not btn then return rolls, (rolling() and "stuck" or "nobutton") end
					gold = readGold(blessingsWindow()) or gold
					if gold < cost then return rolls, "poor" end
				end
				if IsToggled("RollStopOnMatchToggle") and rolls > 0 and rewardState() == "reward" then
					task.wait(0.8)
					if rewardState() == "reward" then return rolls, "match" end
				end
				status(("Rolling... #%d (gold %d)"):format(rolls + 1, gold))
				Gui.Press(Gui.ButtonOf(btn), misses + 1)
				local t0, spentGold = os.clock(), false
				while os.clock() - t0 < 4 do
					task.wait(0.25)
					local g = readGold(blessingsWindow())
					if g and g < gold then spentGold = true break end
				end
				if spentGold then
					rolls += 1
					misses = 0
					task.wait(ROLL_WAIT)
				else
					misses += 1
					if misses >= 3 then return rolls, "stuck" end
					task.wait(1)
				end
			end
			return rolls, "limit"
		end
		local function runBlessing()
			if running then return false end
			Gui.NpcAcquire("Blessing")
			running = true
			local root = Gui.Root()
			local homeCF = root and root.CFrame
			local rolls, result = 0, nil
			local ok, err = pcall(function()
				if not openNpc("Roll", "BlessingStatus", P, function(t)
					return t:match("^%[?e%]?%s*roll") ~= nil or t:match("^%[?e%]?%s*blessing") ~= nil
				end, function() return blessingsWindow() ~= nil end) then
					status("Could not open the Blessings window")
					error("open failed", 0)
				end
				task.wait(0.6)
				rolls, result = rollLoop()
				if result == "match" then
					status(("Done: 3 cards matched after %d roll(s). Auto Roll stopped."):format(rolls))
					SetToggle("AutoRollBlessingToggle", false)
				elseif result == "poor" then
					status(("Done: rolled %d time(s), gold is below 2,000"):format(rolls))
				elseif result == "stuck" then
					status(("Stopped: ROLL did not respond (rolled %d)"):format(rolls))
				elseif result == "nogold" then
					status("Stopped: can't read your gold")
				elseif result == "nobutton" then
					status("Stopped: ROLL button not found")
				else
					status(("Done: rolled %d time(s)"):format(rolls))
				end
			end)
			if not ok and err ~= "open failed" then status("Error: " .. tostring(err)) end
			pcall(closeWindow)
			if Gui.Root() and homeCF then Gui.FlyTo(homeCF) end
			running = false
			Gui.NpcRelease("Blessing")
			return ok, rolls, result
		end

		task.spawn(function()
			local fails, nextTry = 0, 0
			while alive() do
				task.wait(1)
				if not IsToggled("AutoRollBlessingToggle") then
					if not running then status("Off") end
					fails = 0
				elseif not inMatch() or not waveNpc("Roll") then
					status("Waiting for wave mode")
				elseif not running and not KH.State.Refilling and os.clock() >= nextTry and not Gui.NpcCooldown("Blessing") then
					local gold = readGold(nil)
					if gold == nil then
						status("Can't read your gold")
					elseif gold < COST then
						status(("Waiting for gold (%d / %d)"):format(gold, COST))
					else
						local ok, rolls, result = runBlessing()
						if ok and ((rolls or 0) > 0 or result == "poor") then
							fails = 0
							nextTry = os.clock() + 3
						else
							fails += 1
							nextTry = os.clock() + 8
							if fails >= 3 then
								fails = 0
								nextTry = os.clock() + 60
							end
						end
					end
				end
			end
		end)
	end

	----------------------------------------------------------------------------
	-- FPS lock: the executor's own cap function when there is one, otherwise a
	-- built-in limiter (up to 240 FPS). Live FPS shown under the preset.
	----------------------------------------------------------------------------
	do
		local PRESETS = {["15 FPS"] = 15, ["30 FPS"] = 30, ["60 FPS"] = 60, ["120 FPS"] = 120, ["144 FPS"] = 144,
			["240 FPS"] = 240, ["500 FPS"] = 500, ["1000 FPS"] = 1000, ["9999 (Unlimited)"] = 9999}
		local candidates = {setfpscap, set_fps_cap, setfps, syn and syn.set_fps_cap, fluxus and fluxus.set_fps_cap}
		local names = {"setfpscap", "set_fps_cap", "setfps", "syn.set_fps_cap", "fluxus.set_fps_cap"}
		local capFn, capName
		for i = 1, 5 do
			if type(candidates[i]) == "function" then capFn, capName = candidates[i], names[i] break end
		end
		local Fps = {Mode = "off", Measured = 0}
		KH.Fps = Fps
		local softConn
		local function stopSoft()
			if softConn then softConn:Disconnect() softConn = nil end
		end
		local function limit()
			return math.clamp(PRESETS[GetOption("FpsPresetDropdown", "144 FPS")] or 144, 1, 9999)
		end
		local function startSoft(fps)
			stopSoft()
			local frame, last = 1 / fps, os.clock()
			softConn = RunService.RenderStepped:Connect(function()
				if not alive() then stopSoft() return end
				local untilT = last + frame
				while os.clock() < untilT do end
				last = os.clock()
			end)
		end
		local applied = {on = nil, limit = nil}
		function Fps.Apply()
			local on, lim = IsToggled("FpsCapToggle"), limit()
			stopSoft()
			if on then
				if capFn then
					pcall(capFn, lim)
					Fps.Mode = "native"
				elseif lim >= 9999 then
					Fps.Mode = "off"
				elseif lim <= 240 then
					startSoft(lim)
					Fps.Mode = "software"
				else
					Fps.Mode = "unsupported"
				end
			else
				if capFn then pcall(capFn, 9999) end
				Fps.Mode = "off"
			end
			applied.on, applied.limit = on, lim
		end
		function Fps.Off()
			stopSoft()
			if capFn then pcall(capFn, 9999) end
			Fps.Mode = "off"
		end
		local frames, since = 0, os.clock()
		local countConn
		countConn = RunService.RenderStepped:Connect(function()
			if not alive() then countConn:Disconnect() return end
			frames += 1
		end)
		task.spawn(function()
			while alive() do
				task.wait(0.5)
				local now = os.clock()
				if now - since > 0 then Fps.Measured = frames / (now - since) end
				frames, since = 0, now
				-- keep the lock applied (some games reset the cap) and pick up option changes
				local on, lim = IsToggled("FpsCapToggle"), limit()
				if on ~= applied.on or lim ~= applied.limit then
					Fps.Apply()
				elseif on and Fps.Mode == "native" then
					pcall(capFn, lim)
				end
				local note
				if Fps.Mode == "native" then
					note = "Locked at " .. lim .. " FPS using " .. tostring(capName) .. "."
				elseif Fps.Mode == "software" then
					note = "Locked at " .. lim .. " FPS by the built-in limiter (your executor has no FPS cap function)."
				elseif Fps.Mode == "unsupported" then
					note = "Can't lock above 240 FPS: your executor has no FPS cap function."
				else
					note = "Lock is off." .. (capName and (" Executor supports " .. capName .. ".") or " Executor has no FPS cap function (built-in limiter will be used up to 240 FPS).")
				end
				setLabel("FpsStatus", ("Current FPS ( %d )\n%s"):format(math.floor(Fps.Measured + 0.5), note))
			end
		end)
	end

	----------------------------------------------------------------------------
	-- Share config: copy your switches as a short code, or import a code / link /
	-- file into a new config in the list (press Load to use it). Only settings
	-- that differ from the defaults are in a code; webhook URLs and keys never are.
	----------------------------------------------------------------------------
	do
		local PREFIX = "KIRACFG1:"
		local SHARE_FOLDER = CFG_FOLDER .. "/Share"
		local MAX_CODE = 300000
		local B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
		local B64_IDX = {}
		for i = 1, #B64 do B64_IDX[B64:byte(i)] = i - 1 end

		local function status(t) setLabel("ShareStatus", "Share: " .. tostring(t)) end
		local function encode(s)
			local out = {}
			for i = 1, #s, 3 do
				local a, b, c = s:byte(i, i + 2)
				local n = a * 65536 + (b or 0) * 256 + (c or 0)
				local c1, c2 = bit32.rshift(n, 18) % 64, bit32.rshift(n, 12) % 64
				local c3, c4 = bit32.rshift(n, 6) % 64, n % 64
				out[#out + 1] = B64:sub(c1 + 1, c1 + 1) .. B64:sub(c2 + 1, c2 + 1)
					.. (b and B64:sub(c3 + 1, c3 + 1) or "=") .. (c and B64:sub(c4 + 1, c4 + 1) or "=")
			end
			return table.concat(out)
		end
		local function decode(s)
			s = s:gsub("[^%w%+/=]", "")
			local out = {}
			for i = 1, #s, 4 do
				local a, b, c, d = s:byte(i, i + 3)
				local va, vb = B64_IDX[a], B64_IDX[b or 0]
				if not va or not vb then return nil end
				local vc = (c and c ~= 61) and B64_IDX[c] or nil
				local vd = (d and d ~= 61) and B64_IDX[d] or nil
				local n = va * 262144 + vb * 4096 + (vc or 0) * 64 + (vd or 0)
				out[#out + 1] = string.char(bit32.rshift(n, 16) % 256)
				if vc then out[#out + 1] = string.char(bit32.rshift(n, 8) % 256) end
				if vd then out[#out + 1] = string.char(n % 256) end
			end
			return table.concat(out)
		end
		local function same(a, b)
			if type(a) ~= type(b) then return false end
			if type(a) == "table" then
				if #a ~= #b then return false end
				for i = 1, #a do if a[i] ~= b[i] then return false end end
				return true
			end
			return a == b
		end
		-- only the switches that differ from the defaults
		local function diff(values, defaults)
			local out = {}
			for k, v in pairs(values) do
				if not same(v, defaults[k]) then out[k] = v end
			end
			return out
		end

		local function buildCode()
			local snap = KH.SnapshotOptions()
			local defaults = KH.ConfigDefaults or {t = {}, o = {}}
			local name = "Shared"
			pcall(function()
				local sel = Options.SaveManager_ConfigList and Options.SaveManager_ConfigList.Value
				if type(sel) == "string" and sel ~= "" then name = sel end
			end)
			local ok, json = pcall(function()
				return HttpService:JSONEncode({n = name, t = diff(snap.t, defaults.t), o = diff(snap.o, defaults.o)})
			end)
			if not ok then return nil end
			return PREFIX .. encode(json), name
		end

		local function readSource(src)
			src = tostring(src or ""):match("^%s*(.-)%s*$")
			if src == "" then return nil, "Paste a share code, link or file name first" end
			if src:match("^https?://") then
				local ok, body = pcall(function() return game:HttpGet(src) end)
				if not ok or type(body) ~= "string" or body == "" then return nil, "Download failed (check the link)" end
				return body
			end
			if src:find(PREFIX, 1, true) then return src end
			if type(isfile) == "function" and type(readfile) == "function" then
				for _, path in ipairs({SHARE_FOLDER .. "/" .. src, SHARE_FOLDER .. "/" .. src .. ".txt", src}) do
					local okF, exists = pcall(isfile, path)
					if okF and exists then
						local okR, body = pcall(readfile, path)
						if okR and type(body) == "string" then return body end
					end
				end
			end
			return nil, "Not a valid share code, link or file"
		end
		local function parseCode(text)
			text = tostring(text):gsub("%s", "")
			local at = text:find(PREFIX, 1, true)
			if not at then return nil, "Not a Kira Hub share code" end
			text = text:sub(at + #PREFIX)
			if #text > MAX_CODE then return nil, "Code is too large" end
			local json = decode(text)
			if not json or json == "" then return nil, "Code is damaged" end
			local ok, data = pcall(function() return HttpService:JSONDecode(json) end)
			if not ok or type(data) ~= "table" then return nil, "Code is damaged" end
			return data
		end

		local function settingsFolder()
			local base = SaveManager.Folder or (CFG_FOLDER .. "/Configs")
			local folder = base .. "/settings"
			if type(isfolder) == "function" and not isfolder(folder) and isfolder(base) then folder = base end
			if type(SaveManager.SubFolder) == "string" and SaveManager.SubFolder ~= "" then
				folder = folder .. "/" .. SaveManager.SubFolder
			end
			return folder
		end

		-- write a SaveManager config file: current values with the imported ones merged on top
		-- (only keys that exist here with the same type), using SaveManager's own parsers
		local function saveImported(name, data)
			local snap = KH.SnapshotOptions()
			local t, o = {}, {}
			for k, v in pairs(snap.t) do t[k] = v end
			for k, v in pairs(snap.o) do o[k] = v end
			local applied = 0
			for k, v in pairs(type(data.t) == "table" and data.t or {}) do
				if snap.t[k] ~= nil and type(snap.t[k]) == type(v) then t[k] = v applied += 1 end
			end
			for k, v in pairs(type(data.o) == "table" and data.o or {}) do
				if snap.o[k] ~= nil and type(snap.o[k]) == type(v) then o[k] = v applied += 1 end
			end
			if applied == 0 then return nil, "No supported settings found in that code" end

			local objects = {}
			local parser = SaveManager.Parser or {}
			for idx, v in pairs(t) do
				local real = Toggles[idx]
				local fake = setmetatable({Value = v}, {__index = real})
				local okP, entry = pcall(function() return parser.Toggle and parser.Toggle.Save(idx, fake) end)
				objects[#objects + 1] = (okP and entry) or {type = "Toggle", idx = idx, value = v}
			end
			for idx, v in pairs(o) do
				local real = Options[idx]
				local kind = real and real.Type
				local value = v
				if kind == "Dropdown" and real.Multi and type(v) == "table" then
					value = {}
					for _, name2 in ipairs(v) do value[name2] = true end
				end
				local fake = setmetatable({Value = value}, {__index = real})
				local okP, entry = pcall(function() return parser[kind] and parser[kind].Save(idx, fake) end)
				if okP and entry then
					objects[#objects + 1] = entry
				elseif kind == "Slider" then
					objects[#objects + 1] = {type = "Slider", idx = idx, value = tostring(value)}
				elseif kind == "Dropdown" then
					objects[#objects + 1] = {type = "Dropdown", idx = idx, value = value, multi = real.Multi}
				elseif kind == "Input" then
					objects[#objects + 1] = {type = "Input", idx = idx, text = value}
				end
			end

			local folder = settingsFolder()
			local base = tostring(name):gsub("[^%w_%- ]", "_")
			if base == "" then base = "Shared" end
			local final, n = base, 1
			while type(isfile) == "function" and isfile(folder .. "/" .. final .. ".json") do
				n += 1
				final = base .. " (" .. n .. ")"
			end
			local okW = pcall(function()
				if type(makefolder) == "function" and not isfolder(folder) then makefolder(folder) end
				writefile(folder .. "/" .. final .. ".json", HttpService:JSONEncode({objects = objects}))
			end)
			pcall(function()
				local list = SaveManager:RefreshConfigList()
				Options.SaveManager_ConfigList:SetValues(list)
				Options.SaveManager_ConfigList:SetValue(final)
			end)
			return final, nil, okW, applied
		end

		KH.Share = {
			Copy = function()
				local code, name = buildCode()
				if not code then status("Could not build the code") return end
				local saved = false
				if type(writefile) == "function" then
					pcall(function()
						if type(makefolder) == "function" and type(isfolder) == "function" and not isfolder(SHARE_FOLDER) then
							makefolder(SHARE_FOLDER)
						end
						writefile(SHARE_FOLDER .. "/" .. tostring(name):gsub("[^%w_%- ]", "_") .. ".txt", code)
						saved = true
					end)
				end
				local clip = setclipboard or toclipboard or (syn and syn.write_clipboard)
				if type(clip) == "function" then
					pcall(clip, code)
					status(("Copied \"%s\" (%d chars)%s"):format(name, #code,
						#code > 1900 and " - too long for a normal Discord message, send it as a .txt file" or ""))
				elseif saved then
					status("No clipboard - saved to workspace/" .. SHARE_FOLDER .. "/")
				else
					status("No clipboard and can't save files")
				end
			end,
			Import = function(src)
				status("Importing...")
				local ok, final, err, written, applied = pcall(function()
					local text, why = readSource(src)
					if not text then return nil, why end
					local data, why2 = parseCode(text)
					if not data then return nil, why2 end
					return saveImported(type(data.n) == "string" and data.n or "Shared", data)
				end)
				if not ok then status("Import failed") return end
				if not final then status("Failed: " .. tostring(err)) return end
				status(("Imported: %s (%d settings)%s - press Load"):format(final, applied or 0, written == false and " (not saved to file)" or ""))
			end,
		}
	end
end)()

local function ExecuteAutoStartLogic()
	if not IsToggled("AutoStartToggle") then return end
	if not isLobby then return end
	if IsToggled("SkinHunterToggle") then return end
	if IsToggled("AutoTSQuestToggle") then return end
	
	for _, mission in ipairs(ReplicatedStorage.Missions:GetChildren()) do
		if mission:FindFirstChild("Leader") and mission.Leader.Value == player.Name then
			pcall(function() getRemote:InvokeServer("S_Missions", "Leave") end)
		end
	end
	
	local missionType = GetOption("StartTypeDropdown", "Missions")
	local difficulty, mapName, objective
	
	if missionType == "Missions" then
		difficulty = GetOption("MissionDifficultyDropdown", "Hardest")
		mapName = GetOption("MissionMapDropdown", "Shiganshina")
		objective = GetOption("MissionObjectiveDropdown", "Skirmish")
	elseif missionType == "Raids" then
		difficulty = GetOption("RaidDifficultyDropdown", "Hardest")
		mapName = GetOption("RaidMapDropdown", "Trost")
		objective = getgenv().KiraHub.Data.RaidObjectives[mapName] or "Attack Titan"
		mapName = getgenv().KiraHub.Data.RaidMapNames[mapName] or mapName
	elseif missionType == "Waves" then
		difficulty = "Easy"
		mapName = GetOption("WavesMapDropdown", "Trost")
		objective = "Waves"
	end
	
	local created = false
	
	if difficulty == "Hardest" then
		local diffOrder = missionType == "Raids"
			and {"Aberrant", "Severe", "Hard"}
			or {"Aberrant", "Severe", "Hard", "Normal", "Easy"}
		for _, diff in ipairs(diffOrder) do
			if not IsToggled("AutoStartToggle") then break end
			pcall(function()
				getRemote:InvokeServer("S_Missions", "Create", {
					Difficulty = diff,
					Type = missionType == "Raids" and "Raids" or missionType,
					Name = mapName,
					Objective = objective
				})
			end)
			if GetMyMission() then
				created = true
				break
			end
		end
	else
		pcall(function()
			getRemote:InvokeServer("S_Missions", "Create", {
				Difficulty = difficulty,
				Type = missionType == "Raids" and "Raids" or missionType,
				Name = mapName,
				Objective = objective
			})
		end)
		if GetMyMission() then created = true end
	end
	
	if not created then return end
	
	local activeMods = {}
	local uiMods = GetOption("ModifiersDropdown", {})
	if type(uiMods) == "table" then
		for modName, isActive in pairs(uiMods) do
			if isActive then table.insert(activeMods, modName) end
		end
	end
	
	for _, modifier in ipairs(activeMods) do
		pcall(function() getRemote:InvokeServer("S_Missions", "Modify", modifier) end)
	end
	task.wait(0.5)
	
	local startDelay = GetOption("StartDelaySlider", 0)
	if startDelay > 0 then task.wait(startDelay) end
	
	pcall(function() getRemote:InvokeServer("S_Missions", "Start") end)
	task.wait(5)
end

task.spawn(function()
	while true do
		task.wait(1)
		if not getgenv().KiraHubAOTR then break end
		pcall(ExecuteAutoStartLogic)
	end
end)

local function UpdateStateLabels()
	local elapsed = os.time() - getgenv().KiraHub.State.StartTime
	if elapsed < 0 then elapsed = 0 end
	local hours = math.floor(elapsed / 3600)
	local mins = math.floor((elapsed % 3600) / 60)
	local secs = elapsed % 60
	
	pcall(function()
		getgenv().KiraHub.UI.Labels.SessTime:SetText(string.format("Session Time: %02d:%02d:%02d", hours, mins, secs))
		getgenv().KiraHub.UI.Labels.SessGames:SetText("Games Played: " .. tostring(getgenv().KiraHub.State.GamesPlayed))
		getgenv().KiraHub.UI.Labels.SessGold:SetText("Total Gold: " .. tostring(getgenv().KiraHub.State.TotalGold))
		getgenv().KiraHub.UI.Labels.SessGems:SetText("Total Gems: " .. tostring(getgenv().KiraHub.State.TotalGems))
		getgenv().KiraHub.UI.Labels.SessXP:SetText("Total XP: " .. tostring(getgenv().KiraHub.State.TotalXP))
		getgenv().KiraHub.UI.Labels.SessMythical:SetText("Mythical Drops: " .. tostring(getgenv().KiraHub.State.MythicalDrops))
		getgenv().KiraHub.UI.Labels.SessCrashes:SetText("Crashes Detected: " .. tostring(getgenv().KiraHub.State.Crashes))
		
		local hRatio = elapsed / 3600
		local goldRate = hRatio > 0.01 and math.floor(getgenv().KiraHub.State.TotalGold / hRatio) or 0
		local gamesRate = hRatio > 0.01 and math.floor(getgenv().KiraHub.State.GamesPlayed / hRatio) or 0
		local avgGoldVal = getgenv().KiraHub.State.GamesPlayed > 0 and math.floor(getgenv().KiraHub.State.TotalGold / getgenv().KiraHub.State.GamesPlayed) or 0
		
		getgenv().KiraHub.UI.Labels.RateGoldHour:SetText("Gold / Hour: " .. tostring(goldRate))
		getgenv().KiraHub.UI.Labels.RateGamesHour:SetText("Games / Hour: " .. tostring(gamesRate))
		getgenv().KiraHub.UI.Labels.RateAvgGold:SetText("Avg Gold / Game: " .. tostring(avgGoldVal))
		
		local lastFb = getgenv().KiraHub.State.FeedbackCooldown or 0
		local fbElapsed = tick() - lastFb
		if fbElapsed < 3600 then
			local rem = math.ceil(3600 - fbElapsed)
			local fMins = math.floor(rem / 60)
			local fSecs = rem % 60
			getgenv().KiraHub.UI.Labels.FeedbackCooldown:SetText(string.format("Cooldown: %02dm %02ds", fMins, fSecs))
		else
			getgenv().KiraHub.UI.Labels.FeedbackCooldown:SetText("Cooldown: Ready")
		end
	end)
end

local function SaveAllStates()
	pcall(function()
		local states = {
			toggles = {},
			sliders = {},
			dropdowns = {},
			theme = {
				FontFace = "Jura",
				FontWeight = "Medium"
			},			
			inputs = {},
			_version = "v2.9"
		}

		if Toggles then
			for name, t in pairs(Toggles) do
				if type(t.Value) == "boolean" then
					states.toggles[name] = t.Value
				end
			end
		end

		if Options then
			for name, opt in pairs(Options) do
				local val = opt.Value
				local vType = type(val)

				if vType == "number" then
					states.sliders[name] = val
				elseif vType == "string" then
					states.dropdowns[name] = val
				elseif vType == "table" then
					local cleanTable = {}
					local isValid = false
					for k, v in pairs(val) do
						if (type(k) == "string" or type(k) == "number") and type(v) == "boolean" then
							cleanTable[tostring(k)] = v
							isValid = true
						end
					end
					if isValid then
						states.dropdowns[name] = cleanTable
					end
				end
			end
		end

		local encoded = HttpService:JSONEncode(states)
		writefile(AllStatesFile, encoded)
	end)
end

local function LoadAllStates()
	pcall(function()
		if not isfile(AllStatesFile) then return end
		
		local raw = readfile(AllStatesFile)
		if not raw or raw == "" then return end

		local ok, states = pcall(HttpService.JSONDecode, HttpService, raw)
		if not ok or type(states) ~= "table" then return end

		if states.toggles and type(states.toggles) == "table" then
			for name, val in pairs(states.toggles) do
				if Toggles[name] and type(val) == "boolean" then
					pcall(function() Toggles[name]:SetValue(val) end)
				end
			end
		end

		if states.sliders and type(states.sliders) == "table" then
			for name, val in pairs(states.sliders) do
				if Options[name] and type(val) == "number" then
					pcall(function() Options[name]:SetValue(val) end)
				end
			end
		end

		if states.dropdowns and type(states.dropdowns) == "table" then
			for name, val in pairs(states.dropdowns) do
				if Options[name] then
					pcall(function() Options[name]:SetValue(val) end)
				end
			end
		end
	end)
end
